function getId(value) {
    const id = Number(value);

// ADMIN LOGIN
// POST /api/auth/admin/login
// =====================================================

app.post("/api/auth/admin/login", async (req, res) => {
    try {
        const { email, password } = req.body || {};

        // -------------------------------------------------
        // VALIDATION
        // -------------------------------------------------

        if (!email || !password) {
            return res.status(400).json({
                success: false,
                error: "Email and password are required"
            });
        }

        const normalizedEmail = email.trim().toLowerCase();

        console.log("=================================");
        console.log("ADMIN LOGIN REQUEST");
        console.log("Email:", normalizedEmail);
        console.log("=================================");

        // -------------------------------------------------
        // FIND ADMIN
        //
        // This assumes your admin users are stored in:
        // third_party_users
        // -------------------------------------------------

        const {
            data: admin,
            error
        } = await supabase
            .from("third_party_users")
            .select("*")
            .eq("email", normalizedEmail)
            .maybeSingle();

        // -------------------------------------------------
        // DATABASE ERROR
        // -------------------------------------------------

        if (error) {
            console.error("Admin database error:", error);

            return res.status(500).json({
                success: false,
                error: error.message
            });
        }

        // -------------------------------------------------
        // ADMIN NOT FOUND
        // -------------------------------------------------

        if (!admin) {
            console.log("Admin not found:", normalizedEmail);

            return res.status(401).json({
                success: false,
                error: "Invalid Email or Password"
            });
        }

        // -------------------------------------------------
        // CHECK ROLE
        // -------------------------------------------------

        if (
            !admin.role ||
            String(admin.role).trim().toLowerCase() !== "admin"
        ) {
            console.log(
                "User exists but is not admin:",
                admin.role
            );

            return res.status(403).json({
                success: false,
                error: "This account is not an admin account"
            });
        }

        // -------------------------------------------------
        // PASSWORD CHECK
        // -------------------------------------------------

        if (admin.password !== password) {
            console.log("Invalid admin password");

            return res.status(401).json({
                success: false,
                error: "Invalid Email or Password"
            });
        }

        // -------------------------------------------------
        // SUCCESS
        // -------------------------------------------------

        console.log(
            "Admin login successful:",
            normalizedEmail
        );

        return res.status(200).json({
            success: true,

            role: "admin",

            token: `admin-${admin.id}`,

            user_email: admin.email,

            company_name:
                admin.company_name ||
                "Talent Corner HR Services",

            client_id:
                admin.client_id || null,

            user: {
                id: admin.id,

                email: admin.email,

                role: "admin",

                name:
                    admin.full_name ||
                    admin.name ||
                    "Admin"
            }
        });

    } catch (error) {
        console.error(
            "Admin login exception:",
            error
        );

        return res.status(500).json({
            success: false,
            error: "Admin login failed"
        });
    }
});

// UPDATE CLIENT
app.put("/api/clients/:id", async (req, res) => {
    try {
        const id = getId(req.params.id);

        if (!id) {
            return sendError(res, 400, "Invalid client ID");
        }

        const allowedFields = [
            "company_name",
            "gstin",
            "billing_address",
            "state_code",
            "credit_terms",
        ];

        const updates = {};

        for (const field of allowedFields) {
            if (req.body[field] !== undefined) {
                updates[field] = req.body[field];
            }
        }

        if (Object.keys(updates).length === 0) {
            return sendError(
                res,
                400,
                "No valid fields provided"
            );
        }

        const { data, error } = await supabase
            .from("clients")
            .update(updates)
            .eq("id", id)
            .select()
            .single();

        if (error) throw error;

        res.json({
            success: true,
            message: "Client updated successfully",
            data,
        });
    } catch (err) {
        console.error("PUT /api/clients/:id:", err.message);

        sendError(res, 500, err.message);
    }
});

// UPDATE CONTRACT
app.put("/api/contracts/:id", async (req, res) => {
    try {
        const id = getId(req.params.id);

        if (!id) {
            return sendError(res, 400, "Invalid contract ID");
        }

        const allowedFields = [
            "client_id",
            "contract_number",
            "contract_title",
            "billing_model",
            "markup_percentage",
            "per_head_fee",
            "credit_terms",
            "start_date",
            "end_date",
            "contract_status",
        ];

        const updates = {};

        for (const field of allowedFields) {
            if (req.body[field] !== undefined) {
                if (field === "client_id") {
                    const value = getId(req.body[field]);

                    if (!value) {
                        return sendError(
                            res,
                            400,
                            "Invalid client_id"
                        );
                    }

                    updates[field] = value;
                } else if (
                    field === "markup_percentage" ||
                    field === "per_head_fee"
                ) {
                    updates[field] = Number(req.body[field]);
                } else {
                    updates[field] = req.body[field];
                }
            }
        }

        if (Object.keys(updates).length === 0) {
            return sendError(
                res,
                400,
                "No valid fields provided"
            );
        }

        const { data, error } = await supabase
            .from("client_contracts")
            .update(updates)
            .eq("id", id)
            .select()
            .single();

        if (error) throw error;

        res.json({
            success: true,
            message: "Contract updated successfully",
            data,
        });
    } catch (err) {
        console.error("PUT /api/contracts/:id:", err.message);

        sendError(res, 500, err.message);
    }
});

app.get("/api/deployments", async (req, res) => {
    try {
        const { data, error } = await supabase
            .from("deployments")
            .select(`
                    *,
                    candidates (
                        id,
                        full_name,
                        email,
                        phone,
                        designation
                    ),
                    clients (
                        id,
                        company_name
                    ),
                    client_contracts (
                        id,
                        contract_number,
                        contract_title
                    )
                `)
            .order("id", { ascending: false });

        if (error) {
            throw error;
        }

        const formattedData = (data || []).map((deployment) => {

            const payRate = Number(deployment.pay_rate || 0);
            const billRate = Number(deployment.bill_rate || 0);

            // Service charge / margin
            const serviceCharge = billRate - payRate;

            // Service charge percentage based on client bill
            const serviceChargePercentage =
                billRate > 0
                    ? ((serviceCharge / billRate) * 100).toFixed(2)
                    : "0.00";

            return {
                ...deployment,

                // Employee information
                employee_name:
                    deployment.candidates?.full_name || "N/A",

                employee_email:
                    deployment.candidates?.email || "N/A",

                employee_phone:
                    deployment.candidates?.phone || "N/A",

                designation:
                    deployment.candidates?.designation || "N/A",

                // Client information
                client_name:
                    deployment.clients?.company_name || "N/A",

                company_name:
                    deployment.clients?.company_name || "N/A",

                // Contract information
                contract_number:
                    deployment.client_contracts?.contract_number || "N/A",

                contract_title:
                    deployment.client_contracts?.contract_title || "N/A",

                // Financial information
                pay_rate: payRate,

                bill_rate: billRate,

                service_charge: serviceCharge,

                service_charge_percentage:
                    Number(serviceChargePercentage),
            };
        });

        res.json({
            success: true,
            data: formattedData,
        });

    } catch (err) {

        console.error(
            "GET /api/deployments:",
            err.message
        );

        res.status(500).json({
            success: false,
            error: err.message,
        });
    }
});

// ============================================================
// HELPER: GET ACTUAL DAYS IN MONTH
// salaryMonth must be YYYY-MM
// ============================================================

function getDaysInMonth(salaryMonth) {
    if (!salaryMonth) {
        throw new Error("salary_month is required");
    }

    const match = String(salaryMonth).match(/^(\d{4})-(\d{2})$/);

    if (!match) {
        throw new Error(
            "salary_month must be in YYYY-MM format"
        );
    }

    const year = Number(match[1]);
    const month = Number(match[2]);

    if (month < 1 || month > 12) {
        throw new Error("Invalid salary month");
    }

    return new Date(year, month, 0).getDate();
}


// ============================================================
// HELPER: VALIDATE SALARY MONTH
// ============================================================

function validateSalaryMonth(salaryMonth) {
    if (!salaryMonth) {
        throw new Error("salary_month is required");
    }

    const value = String(salaryMonth).trim();

    if (!/^\d{4}-(0[1-9]|1[0-2])$/.test(value)) {
        throw new Error(
            "salary_month must be in YYYY-MM format"
        );
    }

    return value;
}


// ============================================================
// HELPER: CONVERT SALARY MONTH TO DATE
// For payroll_deductions.deduction_month
// ============================================================

function convertSalaryMonthToDate(salaryMonth) {

    const value = validateSalaryMonth(salaryMonth);

    return `${value}-01`;
}


// ============================================================
// 1. GET PAYROLL
// ============================================================

// GET PAYROLL
// GET /api/payroll
//
// Optional:
// ?client_id=1
// ?salary_month=2026-09
// ============================================================

app.get("/api/payroll", async (req, res) => {
    try {

        const clientIdParam = req.query.client_id;
        const salaryMonth = req.query.salary_month;

        // ====================================================
        // 1. FETCH PAYROLL
        // ====================================================

        let query = supabase
            .from("third_party_payroll")
            .select(`
                *,
                deployments (
                    id,
                    candidate_id,
                    client_id,
                    pay_rate,
                    bill_rate,
                    project_name,
                    billing_model,
                    start_date,
                    end_date,
                    status,
                    clients (
                        id,
                        company_name
                    )
                )
            `)
            .order("id", {
                ascending: false
            });

        // ====================================================
        // 2. FILTER BY SALARY MONTH
        // ====================================================

        if (salaryMonth) {

            const validMonth =
                validateSalaryMonth(salaryMonth);

            query = query.eq(
                "salary_month",
                validMonth
            );
        }

        // ====================================================
        // 3. EXECUTE PAYROLL QUERY
        // ====================================================

        const {
            data,
            error
        } = await query;

        if (error) {
            throw error;
        }

        let result = data || [];

        // ====================================================
        // 4. FILTER BY CLIENT
        // ====================================================

        if (clientIdParam) {

            const clientId =
                getId(clientIdParam);

            if (!clientId) {
                return sendError(
                    res,
                    400,
                    "Invalid client_id"
                );
            }

            result = result.filter(
                payroll =>
                    Number(
                        payroll.client_id ??
                        payroll.deployments?.client_id
                    ) === clientId
            );
        }

        // ====================================================
        // 5. GET ATTENDANCE IDS
        // ====================================================

        const attendanceIds = [
            ...new Set(
                result
                    .map(
                        payroll =>
                            payroll.attendance_id
                    )
                    .filter(
                        id =>
                            id !== null &&
                            id !== undefined
                    )
            )
        ];

        let attendanceById = {};

        // ====================================================
        // 6. FETCH ATTENDANCE
        // ====================================================

        if (attendanceIds.length > 0) {

            const {
                data: attendanceRows,
                error: attendanceError
            } = await supabase
                .from("third_party_emp_attendance")
                .select(`
                    id,
                    employee_id,
                    employee_name,
                    billing_month,
                    present_days,
                    absent_days,
                    leave_days,
                    overtime_hours,
                    approval_status
                `)
                .in(
                    "id",
                    attendanceIds
                );

            if (attendanceError) {
                throw attendanceError;
            }

            attendanceById =
                (attendanceRows || [])
                    .reduce(
                        (acc, row) => {

                            acc[row.id] = row;

                            return acc;
                        },
                        {}
                    );
        }

        // ====================================================
        // 7. ONLY USE APPROVED ATTENDANCE
        // ====================================================

        result = result.filter(
            payroll => {

                const attendance =
                    attendanceById[
                    payroll.attendance_id
                    ];

                return (
                    attendance &&
                    String(
                        attendance.approval_status || ""
                    ).toLowerCase() === "approved"
                );
            }
        );

        // ====================================================
        // 8. GET CANDIDATE IDS
        //
        // Candidate ID can come from:
        // payroll.employee_id
        // payroll.employee_ref_id
        // attendance.employee_id
        // deployment.candidate_id
        // ====================================================

        const candidateIds = [
            ...new Set(
                result
                    .map(payroll => {

                        const attendance =
                            attendanceById[
                            payroll.attendance_id
                            ];

                        return (
                            payroll.employee_id ??
                            payroll.employee_ref_id ??
                            attendance?.employee_id ??
                            payroll.deployments?.candidate_id
                        );
                    })
                    .filter(
                        id =>
                            id !== null &&
                            id !== undefined
                    )
            )
        ];

        // ====================================================
        // 9. FETCH CANDIDATES
        // ====================================================

        let candidateById = {};

        if (candidateIds.length > 0) {

            const {
                data: candidateRows,
                error: candidateError
            } = await supabase
                .from("candidates")
                .select(`
                    id,
                    full_name,
                    email
                `)
                .in(
                    "id",
                    candidateIds
                );

            if (candidateError) {
                throw candidateError;
            }

            candidateById =
                (candidateRows || [])
                    .reduce(
                        (acc, candidate) => {

                            acc[candidate.id] =
                                candidate;

                            return acc;
                        },
                        {}
                    );
        }

        // ====================================================
        // 10. FORMAT FINAL PAYROLL DATA
        // ====================================================

        const formattedData =
            result.map(payroll => {

                // ------------------------------------------------
                // DEPLOYMENT
                // ------------------------------------------------

                const deployment =
                    payroll.deployments || null;

                // ------------------------------------------------
                // CLIENT
                // ------------------------------------------------

                const client =
                    deployment?.clients || null;

                // ------------------------------------------------
                // ATTENDANCE
                // ------------------------------------------------

                const attendance =
                    attendanceById[
                    payroll.attendance_id
                    ] || null;

                // ------------------------------------------------
                // CANDIDATE ID
                // ------------------------------------------------

                const candidateId =
                    payroll.employee_id ??
                    payroll.employee_ref_id ??
                    attendance?.employee_id ??
                    deployment?.candidate_id ??
                    null;

                // ------------------------------------------------
                // CANDIDATE
                // ------------------------------------------------

                const candidate =
                    candidateId !== null
                        ? candidateById[candidateId]
                        : null;

                // ------------------------------------------------
                // EMPLOYEE NAME
                // ------------------------------------------------

                const employeeName =
                    candidate?.full_name ||
                    payroll.employee_name ||
                    attendance?.employee_name ||
                    "N/A";

                // ------------------------------------------------
                // EMPLOYEE EMAIL
                // ------------------------------------------------

                const employeeEmail =
                    candidate?.email ||
                    payroll.employee_email ||
                    payroll.email ||
                    "";

                // ------------------------------------------------
                // ATTENDANCE VALUES
                // ------------------------------------------------

                const presentDays =
                    Number(
                        attendance?.present_days || 0
                    );

                const absentDays =
                    Number(
                        attendance?.absent_days || 0
                    );

                const leaveDays =
                    Number(
                        attendance?.leave_days || 0
                    );

                const overtimeHours =
                    Number(
                        attendance?.overtime_hours || 0
                    );

                // ------------------------------------------------
                // LOP
                //
                // Paid leave is NOT LOP.
                // Only absent days are LOP.
                // ------------------------------------------------

                const lopDays =
                    absentDays;

                // ------------------------------------------------
                // TOTAL CALENDAR DAYS
                // ------------------------------------------------

                let totalDays = null;

                if (attendance?.billing_month) {

                    totalDays =
                        getDaysInMonth(
                            attendance.billing_month
                        );

                } else if (payroll.salary_month) {

                    totalDays =
                        getDaysInMonth(
                            payroll.salary_month
                        );
                }

                // ------------------------------------------------
                // TOTAL DEDUCTIONS
                // ------------------------------------------------

                let totalDeductions;

                if (
                    payroll.total_deductions !== null &&
                    payroll.total_deductions !== undefined &&
                    payroll.total_deductions !== ""
                ) {

                    totalDeductions =
                        Number(
                            payroll.total_deductions
                        ) || 0;

                } else {

                    totalDeductions =
                        Number(payroll.pf || 0) +
                        Number(payroll.esic || 0) +
                        Number(payroll.tax || 0) +
                        Number(
                            payroll.professional_tax || 0
                        ) +
                        Number(payroll.lop || 0);
                }

                // =================================================
                // RETURN PAYROLL RECORD
                // =================================================

                return {

                    ...payroll,

                    // ------------------------------------------------
                    // EMPLOYEE / CANDIDATE
                    // ------------------------------------------------

                    employee_id:
                        candidateId,

                    candidate_id:
                        candidateId,

                    employee_name:
                        employeeName,

                    employee_email:
                        employeeEmail,

                    email:
                        employeeEmail,

                    // ------------------------------------------------
                    // CLIENT
                    // ------------------------------------------------

                    client_id:
                        payroll.client_id ??
                        deployment?.client_id ??
                        null,

                    client_name:
                        client?.company_name ||
                        "N/A",

                    company_name:
                        client?.company_name ||
                        "N/A",

                    // ------------------------------------------------
                    // DEPLOYMENT
                    // ------------------------------------------------

                    deployment_id:
                        payroll.deployment_id ??
                        deployment?.id ??
                        null,

                    project_name:
                        deployment?.project_name ||
                        null,

                    pay_rate:
                        deployment?.pay_rate ??
                        null,

                    bill_rate:
                        deployment?.bill_rate ??
                        null,

                    // ------------------------------------------------
                    // ATTENDANCE
                    // ------------------------------------------------

                    present_days:
                        presentDays,

                    absent_days:
                        absentDays,

                    leave_days:
                        leaveDays,

                    overtime_hours:
                        overtimeHours,

                    total_days:
                        totalDays,

                    lop_days:
                        lopDays,

                    // ------------------------------------------------
                    // DEDUCTIONS
                    // ------------------------------------------------

                    total_deductions:
                        totalDeductions
                };
            });

        // ====================================================
        // 11. SEND RESPONSE
        // ====================================================

        return res.json({
            success: true,
            data: formattedData
        });

    }
    catch (err) {

        console.error(
            "GET /api/payroll:",
            err
        );

        return sendError(
            res,
            500,
            err.message
        );
    }
});

// 3. GENERATE PAYROLL
// ============================================================

app.post("/api/payroll/generate", async (req, res) => {
    try {

        const {
            client_id,
            salary_month
        } = req.body;


        // ============================================================
        // 1. VALIDATION
        // ============================================================

        if (
            client_id === undefined ||
            client_id === null ||
            !salary_month
        ) {
            return sendError(
                res,
                400,
                "client_id and salary_month are required"
            );
        }


        const clientId = Number(client_id);


        if (
            !Number.isInteger(clientId) ||
            clientId <= 0
        ) {
            return sendError(
                res,
                400,
                "Invalid client_id"
            );
        }


        let validSalaryMonth;

        try {

            validSalaryMonth =
                validateSalaryMonth(
                    salary_month
                );

        } catch (monthError) {

            return sendError(
                res,
                400,
                monthError.message
            );
        }


        // ============================================================
        // 2. VERIFY CLIENT
        // ============================================================

        const {
            data: client,
            error: clientError
        } = await supabase
            .from("clients")
            .select(`
                id,
                company_name
            `)
            .eq(
                "id",
                clientId
            )
            .maybeSingle();


        if (clientError) {
            throw clientError;
        }


        if (!client) {
            return sendError(
                res,
                404,
                "Client not found"
            );
        }


        // ============================================================
        // 3. GET APPROVED MONTHLY ATTENDANCE
        // ============================================================

        const {
            data: attendanceRecords,
            error: attendanceError
        } = await supabase
            .from("third_party_emp_attendance")
            .select("*")
            .eq(
                "billing_month",
                validSalaryMonth
            )
            .eq(
                "approval_status",
                "Approved"
            );


        if (attendanceError) {
            throw attendanceError;
        }


        // ============================================================
        // 4. NO APPROVED ATTENDANCE
        // ============================================================

        if (
            !attendanceRecords ||
            attendanceRecords.length === 0
        ) {

            return res.json({

                success: true,

                message:
                    "No approved attendance found for this month",

                data: []

            });
        }


        // ============================================================
        // 5. DAYS IN MONTH
        // ============================================================

        const calendarDays =
            getDaysInMonth(
                validSalaryMonth
            );


        // ============================================================
        // 6. SALARY DIVISOR
        //
        // Payroll calculation uses 26 working days.
        // ============================================================

        const salaryDivisor = 26;


        const payrollRecords = [];


        // ============================================================
        // 7. PROCESS EACH ATTENDANCE RECORD
        // ============================================================

        for (
            const attendance of attendanceRecords
        ) {

            // --------------------------------------------------------
            // DEPLOYMENT REQUIRED
            // --------------------------------------------------------

            if (!attendance.deployment_id) {

                console.log(
                    `Skipping ${attendance.employee_name}: deployment_id missing`
                );

                continue;
            }


            // ========================================================
            // 8. GET DEPLOYMENT
            // ========================================================

            const {
                data: deployment,
                error: deploymentError
            } = await supabase
                .from("deployments")
                .select(`
                    *,
                    clients (
                        id,
                        company_name
                    )
                `)
                .eq(
                    "id",
                    attendance.deployment_id
                )
                .eq(
                    "client_id",
                    clientId
                )
                .single();


            if (
                deploymentError ||
                !deployment
            ) {

                console.log(
                    `Skipping ${attendance.employee_name}: deployment does not belong to selected client`
                );

                continue;
            }


            // ========================================================
            // 9. DEPLOYMENT STATUS
            // ========================================================

            if (
                deployment.status &&
                deployment.status !== "Active"
            ) {

                console.log(
                    `Skipping ${attendance.employee_name}: deployment is not Active`
                );

                continue;
            }


            // ========================================================
            // 10. EMPLOYEE ID
            // ========================================================

            const employeeId =
                attendance.employee_id ||
                attendance.candidate_id ||
                deployment.candidate_id;


            if (!employeeId) {

                console.log(
                    `Skipping ${attendance.employee_name}: employee_id missing`
                );

                continue;
            }


            // ========================================================
            // 11. CHECK EXISTING PAYROLL
            // ========================================================

            const {
                data: existingPayroll,
                error: existingPayrollError
            } = await supabase
                .from("third_party_payroll")
                .select(`
                    id,
                    status
                `)
                .eq(
                    "attendance_id",
                    attendance.id
                )
                .eq(
                    "salary_month",
                    validSalaryMonth
                )
                .maybeSingle();


            if (existingPayrollError) {
                throw existingPayrollError;
            }


            if (existingPayroll) {

                console.log(
                    `Skipping ${attendance.employee_name}: payroll already exists`
                );

                continue;
            }


            // ========================================================
            // 12. PAY RATE
            // ========================================================

            const payRate =
                Number(
                    deployment.pay_rate || 0
                );


            if (payRate <= 0) {

                console.log(
                    `Skipping ${attendance.employee_name}: invalid pay rate`
                );

                continue;
            }


            // ========================================================
            // 13. ATTENDANCE VALUES
            // ========================================================

            const presentDays =
                Number(
                    attendance.present_days || 0
                );

            const absentDays =
                Number(
                    attendance.absent_days || 0
                );

            const leaveDays =
                Number(
                    attendance.leave_days || 0
                );

            const halfDays =
                Number(
                    attendance.half_days || 0
                );

            const overtimeHours =
                Number(
                    attendance.overtime_hours || 0
                );


            // ========================================================
            // 14. LOP DAYS
            //
            // Leave is treated as PAID.
            // Therefore only absent days are LOP.
            // ========================================================

            const lopDays =
                Math.max(
                    0,
                    absentDays
                );


            // ========================================================
            // 15. BASIC SALARY
            // ========================================================

            const basicSalary =
                payRate;


            // ========================================================
            // 16. ALLOWANCES
            // ========================================================

            const allowances = 0;


            // ========================================================
            // 17. BONUS
            // ========================================================

            const bonus = 0;


            // ========================================================
            // 18. DAILY BASIC RATE
            // ========================================================

            const dailyBasicRate =
                basicSalary /
                salaryDivisor;


            // ========================================================
            // 19. LOP DEDUCTION
            // ========================================================

            const lopBasicDeduction =
                Math.round(
                    dailyBasicRate *
                    lopDays
                );


            // ========================================================
            // 20. EARNED BASIC
            // ========================================================

            const earnedBasic =
                Math.max(
                    0,
                    basicSalary -
                    lopBasicDeduction
                );


            // ========================================================
            // 21. OVERTIME RATE
            //
            // Normal hourly rate:
            //
            // Basic / 26 / 8
            //
            // Overtime = 1.5x
            // ========================================================

            const normalHourlyRate =
                basicSalary /
                salaryDivisor /
                8;


            const overtimeRate =
                normalHourlyRate *
                1.5;


            const overtimeAmount =
                Math.round(
                    overtimeHours *
                    overtimeRate
                );


            // ========================================================
            // 22. GROSS SALARY
            // ========================================================

            const grossSalary =
                earnedBasic +
                allowances +
                overtimeAmount +
                bonus;


            // ========================================================
            // 23. PF WAGE
            //
            // PF wage capped at ₹15,000.
            // ========================================================

            const pfBase =
                Math.min(
                    earnedBasic,
                    15000
                );


            // ========================================================
            // 24. EMPLOYEE PF
            //
            // 12%
            // ========================================================

            const employeePF =
                Math.round(
                    pfBase *
                    0.12
                );


            // ========================================================
            // 25. EMPLOYEE ESIC
            //
            // 0.75% when gross <= ₹21,000
            // ========================================================

            let employeeESIC = 0;


            if (
                grossSalary <= 21000
            ) {

                employeeESIC =
                    Math.round(
                        grossSalary *
                        0.0075
                    );
            }


            // ========================================================
            // 26. PROFESSIONAL TAX
            // ========================================================

            const professionalTax =
                200;


            // ========================================================
            // 27. TDS
            // ========================================================

            const taxTds = 0;


            // ========================================================
            // 28. TOTAL EMPLOYEE DEDUCTIONS
            // ========================================================

            const totalDeductions =
                employeePF +
                employeeESIC +
                taxTds +
                professionalTax +
                lopBasicDeduction;


            // ========================================================
            // 29. NET SALARY
            // ========================================================

            const netSalary =
                Math.max(
                    0,
                    grossSalary -
                    totalDeductions
                );


            // ========================================================
            // 30. EMPLOYER PF
            //
            // Employer PF = 12%
            // Same PF wage base and cap.
            // ========================================================

            const employerPF =
                Math.round(
                    pfBase *
                    0.12
                );


            // ========================================================
            // 31. EMPLOYER ESIC
            //
            // 3.25% when gross <= ₹21,000
            // ========================================================

            let employerESIC = 0;


            if (
                grossSalary <= 21000
            ) {

                employerESIC =
                    Math.round(
                        grossSalary *
                        0.0325
                    );
            }


            // ========================================================
            // 32. TOTAL EMPLOYER CONTRIBUTION
            // ========================================================

            const totalEmployerContribution =
                employerPF +
                employerESIC;


            // ========================================================
            // 33. TOTAL EMPLOYER COST
            //
            // Gross salary + employer contribution
            // ========================================================

            const totalEmployerCost =
                grossSalary +
                totalEmployerContribution;


            // ========================================================
            // 34. PAYROLL RECORD
            // ========================================================

            payrollRecords.push({

                // ----------------------------------------------------
                // EMPLOYEE
                // ----------------------------------------------------

                employee_id:
                    employeeId,

                employee_ref_id:
                    employeeId,

                employee_name:
                    attendance.employee_name ||
                    deployment.employee_name ||
                    "Employee",


                // ----------------------------------------------------
                // CLIENT
                // ----------------------------------------------------

                client_id:
                    deployment.client_id ||
                    clientId,


                // ----------------------------------------------------
                // DEPLOYMENT
                // ----------------------------------------------------

                deployment_id:
                    deployment.id,


                // ----------------------------------------------------
                // ATTENDANCE
                // ----------------------------------------------------

                attendance_id:
                    attendance.id,

                total_days:
                    Number(
                        attendance.total_days ??
                        calendarDays
                    ),

                present_days:
                    Number(
                        presentDays.toFixed(2)
                    ),

                absent_days:
                    Number(
                        absentDays.toFixed(2)
                    ),

                leave_days:
                    Number(
                        leaveDays.toFixed(2)
                    ),

                half_days:
                    Number(
                        halfDays.toFixed(2)
                    ),

                lop_days:
                    Number(
                        lopDays.toFixed(2)
                    ),

                payable_days:
                    Number(
                        Math.max(
                            0,
                            presentDays +
                            leaveDays +
                            (halfDays * 0.5) -
                            lopDays
                        ).toFixed(2)
                    ),

                overtime_hours:
                    Number(
                        overtimeHours.toFixed(2)
                    ),


                // ----------------------------------------------------
                // SALARY MONTH
                // ----------------------------------------------------

                salary_month:
                    validSalaryMonth,


                // ----------------------------------------------------
                // EARNINGS
                // ----------------------------------------------------

                basic_salary:
                    Number(
                        earnedBasic.toFixed(2)
                    ),

                allowances:
                    Number(
                        allowances.toFixed(2)
                    ),

                overtime:
                    Number(
                        overtimeAmount.toFixed(2)
                    ),

                overtime_amount:
                    Number(
                        overtimeAmount.toFixed(2)
                    ),

                bonus:
                    Number(
                        bonus.toFixed(2)
                    ),

                gross_salary:
                    Number(
                        grossSalary.toFixed(2)
                    ),


                // ----------------------------------------------------
                // EMPLOYEE DEDUCTIONS
                // ----------------------------------------------------

                pf:
                    Number(
                        employeePF.toFixed(2)
                    ),

                employee_pf:
                    Number(
                        employeePF.toFixed(2)
                    ),

                esic:
                    Number(
                        employeeESIC.toFixed(2)
                    ),

                employee_esic:
                    Number(
                        employeeESIC.toFixed(2)
                    ),

                tax:
                    Number(
                        taxTds.toFixed(2)
                    ),

                tds:
                    Number(
                        taxTds.toFixed(2)
                    ),

                professional_tax:
                    Number(
                        professionalTax.toFixed(2)
                    ),

                lop:
                    Number(
                        lopBasicDeduction.toFixed(2)
                    ),

                lop_deduction:
                    Number(
                        lopBasicDeduction.toFixed(2)
                    ),

                total_deductions:
                    Number(
                        totalDeductions.toFixed(2)
                    ),

                net_salary:
                    Number(
                        netSalary.toFixed(2)
                    ),


                // ----------------------------------------------------
                // EMPLOYER CONTRIBUTIONS
                // ----------------------------------------------------

                employer_pf:
                    Number(
                        employerPF.toFixed(2)
                    ),

                employer_esic:
                    Number(
                        employerESIC.toFixed(2)
                    ),

                total_employer_contribution:
                    Number(
                        totalEmployerContribution.toFixed(2)
                    ),

                total_employer_cost:
                    Number(
                        totalEmployerCost.toFixed(2)
                    ),


                // ----------------------------------------------------
                // BANK DETAILS
                // ----------------------------------------------------

                bank_name:
                    null,

                account_number:
                    null,

                ifsc_code:
                    null,


                // ----------------------------------------------------
                // STATUS
                // ----------------------------------------------------

                status:
                    "Pending"
            });
        }


        // ============================================================
        // 35. NOTHING TO INSERT
        // ============================================================

        if (
            payrollRecords.length === 0
        ) {

            return res.json({

                success: true,

                message:
                    "No new payroll records were generated",

                data: []

            });
        }


        // ============================================================
        // 36. INSERT PAYROLL
        // ============================================================

        const {
            data,
            error
        } = await supabase
            .from("third_party_payroll")
            .insert(
                payrollRecords
            )
            .select();


        if (error) {
            throw error;
        }


        // ============================================================
        // 37. RESPONSE
        // ============================================================

        return res.status(201).json({

            success: true,

            message:
                `Payroll generated successfully for ${validSalaryMonth}`,

            calendar_days:
                calendarDays,

            salary_divisor:
                salaryDivisor,

            client_id:
                clientId,

            client_name:
                client.company_name,

            records_generated:
                data?.length || 0,

            data

        });

    } catch (err) {

        console.error(
            "POST /api/payroll/generate:",
            err
        );

        return sendError(
            res,
            500,
            err.message ||
            "Payroll generation failed"
        );
    }
});


// ============================================================
// 4. UPDATE PAYROLL STATUS
//
// Pending -> Approved -> Locked
// Locked -> NO CHANGE

app.patch(
    "/api/payroll/:id/status",
    async (req, res) => {

        try {

            const id = getId(req.params.id);
            const { status } = req.body;

            // ============================================================
            // 1. VALIDATION
            // ============================================================

            if (!id) {
                return sendError(
                    res,
                    400,
                    "Invalid payroll ID"
                );
            }

            if (!status) {
                return sendError(
                    res,
                    400,
                    "status is required"
                );
            }

            const allowedStatuses = [
                "Pending",
                "Approved",
                "Locked"
            ];

            if (!allowedStatuses.includes(status)) {
                return sendError(
                    res,
                    400,
                    "Invalid payroll status"
                );
            }

            // ============================================================
            // 2. GET PAYROLL
            // ============================================================

            const {
                data: payroll,
                error: payrollError
            } = await supabase
                .from("third_party_payroll")
                .select(`
                    id,
                    employee_name,
                    employee_ref_id,
                    attendance_id,
                    salary_month,

                    basic_salary,
                    allowances,
                    overtime,
                    bonus,
                    gross_salary,

                    pf,
                    esic,
                    tax,
                    professional_tax,
                    lop,
                    total_deductions,
                    net_salary,

                    employer_pf,
                    employer_esic,
                    total_employer_contribution,
                    total_employer_cost,

                    status,
                    deployment_id,
                    client_id
                `)
                .eq("id", id)
                .maybeSingle();

            if (payrollError) {
                throw payrollError;
            }

            if (!payroll) {
                return sendError(
                    res,
                    404,
                    "Payroll record not found"
                );
            }

            // ============================================================
            // 3. CURRENT STATUS
            // ============================================================

            const currentStatus =
                payroll.status || "Pending";

            // ============================================================
            // 4. LOCKED CANNOT CHANGE
            // ============================================================

            if (currentStatus === "Locked") {
                return sendError(
                    res,
                    400,
                    "Cannot modify a Locked payroll record"
                );
            }

            // ============================================================
            // 5. STATUS FLOW
            // ============================================================

            if (
                currentStatus === "Pending" &&
                status !== "Approved"
            ) {
                return sendError(
                    res,
                    400,
                    "Pending payroll can only be changed to Approved"
                );
            }

            if (
                currentStatus === "Approved" &&
                status !== "Locked"
            ) {
                return sendError(
                    res,
                    400,
                    "Approved payroll can only be changed to Locked"
                );
            }

            // ============================================================
            // 6. VERIFY ATTENDANCE BEFORE APPROVING
            // ============================================================

            let attendance = null;

            if (status === "Approved") {

                if (!payroll.attendance_id) {
                    return sendError(
                        res,
                        400,
                        "Payroll does not have a linked attendance record"
                    );
                }

                const {
                    data: attendanceRow,
                    error: attendanceError
                } = await supabase
                    .from("third_party_emp_attendance")
                    .select(`
                        id,
                        employee_id,
                        employee_name,
                        billing_month,
                        present_days,
                        absent_days,
                        leave_days,
                        half_days,
                        overtime_hours,
                        approval_status
                    `)
                    .eq(
                        "id",
                        payroll.attendance_id
                    )
                    .maybeSingle();

                if (attendanceError) {
                    throw attendanceError;
                }

                if (!attendanceRow) {
                    return sendError(
                        res,
                        400,
                        "Linked attendance record not found"
                    );
                }

                if (
                    attendanceRow.approval_status !==
                    "Approved"
                ) {
                    return sendError(
                        res,
                        400,
                        "Payroll cannot be approved because attendance is not approved"
                    );
                }

                attendance = attendanceRow;
            }

            // ============================================================
            // 7. RECALCULATE PAYROLL
            //
            // This happens when Pending -> Approved.
            // ============================================================

            let payrollToUpdate = {
                status
            };

            if (
                currentStatus === "Pending" &&
                status === "Approved"
            ) {

                // --------------------------------------------------------
                // Convert values to numbers
                // --------------------------------------------------------

                const grossSalary =
                    Number(payroll.gross_salary || 0);

                const employeePF =
                    Number(payroll.pf || 0);

                const employeeESIC =
                    Number(payroll.esic || 0);

                const taxTds =
                    Number(payroll.tax || 0);

                const professionalTax =
                    Number(payroll.professional_tax || 0);

                const lopDeduction =
                    Number(payroll.lop || 0);

                // --------------------------------------------------------
                // TOTAL EMPLOYEE DEDUCTIONS
                // --------------------------------------------------------

                const totalDeductions =
                    employeePF +
                    employeeESIC +
                    taxTds +
                    professionalTax +
                    lopDeduction;

                // --------------------------------------------------------
                // NET SALARY
                // --------------------------------------------------------

                const netSalary =
                    Math.max(
                        0,
                        grossSalary -
                        totalDeductions
                    );

                // --------------------------------------------------------
                // EMPLOYER PF
                // --------------------------------------------------------

                const employerPF =
                    Number(
                        payroll.employer_pf || 0
                    );

                // --------------------------------------------------------
                // EMPLOYER ESIC
                // --------------------------------------------------------

                const employerESIC =
                    Number(
                        payroll.employer_esic || 0
                    );

                // --------------------------------------------------------
                // TOTAL EMPLOYER CONTRIBUTION
                // --------------------------------------------------------

                const totalEmployerContribution =
                    employerPF +
                    employerESIC;

                // --------------------------------------------------------
                // TOTAL EMPLOYER COST
                // --------------------------------------------------------

                const totalEmployerCost =
                    grossSalary +
                    totalEmployerContribution;

                // --------------------------------------------------------
                // UPDATE ALL PROCESSED PAYROLL VALUES
                // --------------------------------------------------------

                payrollToUpdate = {

                    status: "Approved",

                    total_deductions:
                        Number(
                            totalDeductions.toFixed(2)
                        ),

                    net_salary:
                        Number(
                            netSalary.toFixed(2)
                        ),

                    employer_pf:
                        Number(
                            employerPF.toFixed(2)
                        ),

                    employer_esic:
                        Number(
                            employerESIC.toFixed(2)
                        ),

                    total_employer_contribution:
                        Number(
                            totalEmployerContribution.toFixed(2)
                        ),

                    total_employer_cost:
                        Number(
                            totalEmployerCost.toFixed(2)
                        )
                };
            }

            // ============================================================
            // 8. UPDATE PAYROLL
            // ============================================================

            const {
                data: updatedPayroll,
                error: updateError
            } = await supabase
                .from("third_party_payroll")
                .update(payrollToUpdate)
                .eq("id", id)
                .select()
                .single();

            if (updateError) {
                throw updateError;
            }

            // ============================================================
            // 9. CREATE DEDUCTION RECORD
            //
            // ONLY Pending -> Approved
            // ============================================================

            if (
                currentStatus === "Pending" &&
                status === "Approved"
            ) {

                // --------------------------------------------------------
                // Check if deduction already exists
                // --------------------------------------------------------

                const {
                    data: existingDeduction,
                    error: existingDeductionError
                } = await supabase
                    .from("deductions")
                    .select("id")
                    .eq(
                        "payroll_id",
                        payroll.id
                    )
                    .maybeSingle();

                if (existingDeductionError) {
                    throw existingDeductionError;
                }

                // --------------------------------------------------------
                // Create deduction record
                // --------------------------------------------------------

                if (!existingDeduction) {

                    const deductionMonth =
                        convertSalaryMonthToDate(
                            payroll.salary_month
                        );

                    const lopDays =
                        attendance
                            ? Number(
                                attendance.absent_days || 0
                            )
                            : Number(
                                payroll.lop || 0
                            );

                    const {
                        data: deduction,
                        error: deductionError
                    } = await supabase
                        .from("deductions")
                        .insert({

                            payroll_id:
                                payroll.id,

                            candidate_id:
                                payroll.employee_ref_id,

                            deduction_month:
                                deductionMonth,

                            lop_days:
                                lopDays,

                            lop_deduction:
                                Number(
                                    payroll.lop || 0
                                ),

                            employee_pf:
                                Number(
                                    payroll.pf || 0
                                ),

                            employee_esic:
                                Number(
                                    payroll.esic || 0
                                ),

                            tax_tds:
                                Number(
                                    payroll.tax || 0
                                ),

                            professional_tax:
                                Number(
                                    payroll.professional_tax || 0
                                ),

                            other_deductions:
                                0
                        })
                        .select()
                        .single();

                    if (deductionError) {
                        throw deductionError;
                    }

                    return res.json({

                        success: true,

                        message:
                            "Payroll approved and deduction record created successfully",

                        payroll:
                            updatedPayroll,

                        deduction
                    });
                }
            }

            // ============================================================
            // 10. NORMAL RESPONSE
            // ============================================================

            return res.json({

                success: true,

                message:
                    `Payroll status changed from ${currentStatus} to ${status}`,

                data:
                    updatedPayroll
            });

        } catch (err) {

            console.error(
                "PATCH /api/payroll/:id/status:",
                err
            );

            return sendError(
                res,
                500,
                err.message ||
                "Failed to update payroll status"
            );
        }
    }
);

app.put(
    "/api/users/:id",
    async (req, res) => {
        try {
            const id = getId(
                req.params.id
            );

            if (!id) {
                return sendError(
                    res,
                    400,
                    "Invalid user ID"
                );
            }

            const allowedFields = [
                "email",
                "password",
                "company_name",
                "role",
                "client_id",
            ];

            const updates = {};

            for (const field of allowedFields) {
                if (
                    req.body[field] !==
                    undefined
                ) {
                    if (
                        field ===
                        "client_id"
                    ) {
                        updates[field] =
                            req.body[field] ===
                                null
                                ? null
                                : Number(
                                    req.body[
                                    field
                                    ]
                                );
                    } else {
                        updates[field] =
                            req.body[field];
                    }
                }
            }

            if (
                Object.keys(updates)
                    .length === 0
            ) {
                return sendError(
                    res,
                    400,
                    "No valid fields provided"
                );
            }

            const { data, error } =
                await supabase
                    .from(
                        "third_party_users"
                    )
                    .update(updates)
                    .eq("id", id)
                    .select(
                        "id, email, company_name, role, created_at, client_id"
                    )
                    .single();

            if (error) throw error;

            res.json({
                success: true,
                message:
                    "User updated successfully",
                data,
            });
        } catch (err) {
            console.error(
                "PUT /api/users/:id:",
                err.message
            );

            sendError(res, 500, err.message);
        }
    }
);

app.get('/api/attendance', async (req, res) => {
    try {
        console.log("GET /api/attendance");

        // 1. Get attendance records
        const {
            data: attendance,
            error: attendanceError
        } = await supabase
            .from('third_party_emp_attendance')
            .select('*')
            .order('id', { ascending: true });

        if (attendanceError) {
            console.error(
                "Attendance table error:",
                attendanceError
            );

            throw attendanceError;
        }

        if (!attendance || attendance.length === 0) {
            return res.json({
                success: true,
                data: []
            });
        }

        // ------------------------------------------
        // 2. Get deployment IDs
        // ------------------------------------------

        const deploymentIds = [
            ...new Set(
                attendance
                    .map(row => row.deployment_id)
                    .filter(Boolean)
            )
        ];

        // ------------------------------------------
        // 3. Get deployments
        // ------------------------------------------

        const {
            data: deployments,
            error: deploymentError
        } = await supabase
            .from('deployments')
            .select(`
            id,
            candidate_id,
            client_id,
            project_name
        `)
            .in('id', deploymentIds);

        if (deploymentError) {
            console.error(
                "Deployment error:",
                deploymentError
            );

            throw deploymentError;
        }

        // ------------------------------------------
        // 4. Get candidate IDs
        // ------------------------------------------

        const candidateIds = [
            ...new Set(
                (deployments || [])
                    .map(d => d.candidate_id)
                    .filter(Boolean)
            )
        ];

        // ------------------------------------------
        // 5. Get candidates
        // ------------------------------------------

        let candidates = [];

        if (candidateIds.length > 0) {
            const {
                data,
                error
            } = await supabase
                .from('candidates')
                .select(`
            id,
            full_name
            `)
                .in('id', candidateIds);

            if (error) {
                console.error(
                    "Candidates error:",
                    error
                );

                throw error;
            }

            candidates = data || [];
        }

        // ------------------------------------------
        // 6. Get client IDs
        // ------------------------------------------

        const clientIds = [
            ...new Set(
                (deployments || [])
                    .map(d => d.client_id)
                    .filter(Boolean)
            )
        ];

        // ------------------------------------------
        // 7. Get clients
        // ------------------------------------------

        let clients = [];

        if (clientIds.length > 0) {
            const {
                data,
                error
            } = await supabase
                .from('clients')
                .select(`
            id,
            company_name
            `)
                .in('id', clientIds);

            if (error) {
                console.error(
                    "Clients error:",
                    error
                );

                throw error;
            }

            clients = data || [];
        }

        // ------------------------------------------
        // 8. Create lookup maps
        // ------------------------------------------

        const deploymentMap = new Map(
            (deployments || []).map(d => [
                d.id,
                d
            ])
        );

        const candidateMap = new Map(
            candidates.map(c => [
                c.id,
                c
            ])
        );

        const clientMap = new Map(
            clients.map(c => [
                c.id,
                c
            ])
        );

        // ------------------------------------------
        // 9. Build Admin response
        // ------------------------------------------

        const formattedRecords = attendance.map(
            record => {

                const deployment =
                    deploymentMap.get(
                        record.deployment_id
                    );

                const candidate =
                    deployment
                        ? candidateMap.get(
                            deployment.candidate_id
                        )
                        : null;

                const client =
                    deployment
                        ? clientMap.get(
                            deployment.client_id
                        )
                        : null;

                return {

                    // Attendance
                    id: record.id,

                    billing_month:
                        record.billing_month,

                    total_days:
                        record.present_days +
                        record.absent_days +
                        record.leave_days,

                    present_days:
                        record.present_days,

                    absent_days:
                        record.absent_days,

                    leave_days:
                        record.leave_days,

                    overtime_hours:
                        Number(
                            record.overtime_hours || 0
                        ),

                    status:
                        record.approval_status ||
                        record.status ||
                        "Pending",

                    approval_status:
                        record.approval_status,

                    created_at:
                        record.created_at,

                    // Employee from candidates
                    employee_ref_id:
                        deployment?.candidate_id || null,

                    employee_id:
                        candidate
                            ? `EMP-${String(
                                candidate.id
                            ).padStart(3, '0')}`
                            : null,

                    employee_name:
                        candidate?.full_name ||
                        record.employee_name ||
                        "N/A",

                    // Client from deployments → clients
                    client_id:
                        deployment?.client_id || null,

                    clientName:
                        client?.company_name ||
                        "N/A",

                    // Deployment
                    deployment_id:
                        record.deployment_id,

                    project_name:
                        deployment?.project_name ||
                        null
                };
            }
        );

        console.log(
            `Found ${formattedRecords.length} attendance records`
        );

        console.log(
            "Formatted attendance:",
            formattedRecords
        );

        return res.json({
            success: true,
            data: formattedRecords
        });

    } catch (err) {

        console.error(
            "GET /api/attendance error:",
            err
        );

        return res.status(500).json({
            success: false,
            error: "Failed to fetch attendance records"
        });
    }
});
async function getClientBillingData(clientId, billingMonth) {

    // -----------------------------------------------------
    // 1. GET ACTIVE CONTRACT
    // -----------------------------------------------------

    let contractQuery = supabase
        .from("client_contracts")
        .select(`
                id,
                client_id,
                contract_number,
                contract_title,
                billing_model,
                markup_percentage,
                per_head_fee,
                credit_terms,
                start_date,
                end_date,
                contract_status
            `)
        .eq("client_id", clientId)
        .eq("contract_status", "Active")
        .order("id", { ascending: false })
        .limit(1);

    const { data: contracts, error: contractError } =
        await contractQuery;

    if (contractError) {
        throw contractError;
    }

    if (!contracts || contracts.length === 0) {
        return {
            contract: null,
            records: [],
        };
    }

    const contract = contracts[0];

    app.get('/api/attendance', async (req, res) => {
        try {
            const { client_id } = req.query;

            let query = supabase
                .from('third_party_attendance_approval')
                .select(`
            id,
            billing_month,
            total_days,
            present_days,
            lop_days,
            ot_hours,
            status,
            created_at,
            client_id,
            employee_ref_id,
            candidates:employee_ref_id (
            id,
            full_name,
            email,
            phone,
            employment_status
            ),
            clients:client_id (
            id,
            company_name
            )
        `)
                .order('id', { ascending: true });

            if (
                client_id &&
                client_id !== 'null' &&
                client_id !== 'undefined' &&
                client_id !== ''
            ) {
                const numericClientId = Number(client_id);

                if (Number.isNaN(numericClientId)) {
                    return res.status(400).json({
                        success: false,
                        error: 'Invalid client_id'
                    });
                }

                query = query.eq('client_id', numericClientId);
            }

            const { data, error } = await query;

            if (error) {
                console.error('Supabase attendance error:', error);
                throw error;
            }

            const formattedData = (data || []).map(record => ({
                id: record.id,

                employee_ref_id: record.employee_ref_id,

                employee_name: record.candidates?.full_name || 'N/A',

                employee_email: record.candidates?.email || 'N/A',

                employee_id: `EMP-${String(record.employee_ref_id).padStart(3, '0')}`,

                role: 'N/A',

                billing_month: record.billing_month,

                total_days: record.total_days,

                present_days: record.present_days,

                lop_days: record.lop_days,

                ot_hours: record.ot_hours,

                status: record.status,

                client_id: record.client_id,

                client_name: record.clients?.company_name || 'N/A',

                created_at: record.created_at
            }));

            console.log('Formatted attendance:', formattedData);

            res.json({
                success: true,
                data: formattedData
            });

        } catch (err) {
            console.error('GET /api/attendance:', err.message);

            res.status(500).json({
                success: false,
                error: err.message
            });
        }
    });


    // -----------------------------------------------------
    // 2. GET ACTIVE DEPLOYMENTS FOR CLIENT
    // -----------------------------------------------------

    const { data: deployments, error: deploymentError } =
        await supabase
            .from("deployments")
            .select(`
                    id,
                    candidate_id,
                    client_id,
                    contract_id,
                    project_name,
                    pay_rate,
                    bill_rate,
                    billing_model,
                    start_date,
                    end_date,
                    status,
                    candidates (
                        id,
                        full_name,
                        email,
                        designation
                    )
                `)
            .eq("client_id", clientId)
            .eq("status", "Active");

    if (deploymentError) {
        throw deploymentError;
    }

    if (!deployments || deployments.length === 0) {
        return {
            contract,
            records: [],
        };
    }

    // -----------------------------------------------------
    // 3. KEEP ONLY DEPLOYMENTS BELONGING TO CONTRACT
    // -----------------------------------------------------

    const contractDeployments = deployments.filter(
        (deployment) =>
            Number(deployment.contract_id) ===
            Number(contract.id)
    );

    if (contractDeployments.length === 0) {
        return {
            contract,
            records: [],
        };
    }

    // -----------------------------------------------------
    // 4. GET APPROVED EMPLOYEE ATTENDANCE
    // -----------------------------------------------------

    const deploymentIds = contractDeployments.map(
        (deployment) => deployment.id
    );

    let attendanceQuery = supabase
        .from("third_party_emp_attendance")
        .select(`
                id,
                employee_name,
                client_id,
                billing_month,
                status,
                present_days,
                absent_days,
                leave_days,
                overtime_hours,
                approval_status,
                approval_record_id,
                deployment_id
            `)
        .eq("client_id", clientId)
        .eq("approval_status", "Approved")
        .in("deployment_id", deploymentIds);

    if (billingMonth) {
        attendanceQuery = attendanceQuery.eq(
            "billing_month",
            billingMonth
        );
    }

    const {
        data: attendanceRecords,
        error: attendanceError,
    } = await attendanceQuery;

    if (attendanceError) {
        throw attendanceError;
    }

    // -----------------------------------------------------
    // 5. CREATE DEPLOYMENT LOOKUP
    // -----------------------------------------------------

    const deploymentMap = {};

    for (const deployment of contractDeployments) {
        deploymentMap[deployment.id] = deployment;
    }

    // -----------------------------------------------------
    // 6. CALCULATE BILLING
    // -----------------------------------------------------

    const records = (attendanceRecords || []).map(
        (attendance) => {

            const deployment =
                deploymentMap[attendance.deployment_id];

            if (!deployment) {
                return null;
            }

            const presentDays =
                Number(attendance.present_days || 0);

            const absentDays =
                Number(attendance.absent_days || 0);

            const leaveDays =
                Number(attendance.leave_days || 0);

            const overtimeHours =
                Number(attendance.overtime_hours || 0);

            const totalBillableDays =
                presentDays;

            const payRate =
                Number(deployment.pay_rate || 0);

            const billRate =
                Number(deployment.bill_rate || 0);

            // -------------------------------------------------
            // BILLING CALCULATION
            // -------------------------------------------------

            let basicBilling = 0;
            let serviceCharge = 0;
            let billingAmount = 0;

            const billingModel =
                contract.billing_model || "Percentage Markup";

            // -------------------------------------------------
            // PERCENTAGE MARKUP
            // -------------------------------------------------

            if (
                billingModel ===
                "Percentage Markup"
            ) {

                // Employee cost for worked days
                const dailyPayRate =
                    payRate / 26;

                const employeeCost =
                    dailyPayRate *
                    totalBillableDays;

                const markupPercentage =
                    Number(
                        contract.markup_percentage || 0
                    );

                serviceCharge =
                    employeeCost *
                    (markupPercentage / 100);

                basicBilling =
                    employeeCost;

                billingAmount =
                    basicBilling +
                    serviceCharge;
            }

            // -------------------------------------------------
            // PER HEAD FEE
            // -------------------------------------------------

            else if (
                billingModel ===
                "Per Head"
            ) {

                const perHeadFee =
                    Number(
                        contract.per_head_fee || 0
                    );

                basicBilling =
                    perHeadFee;

                serviceCharge = 0;

                billingAmount =
                    perHeadFee;
            }

            // -------------------------------------------------
            // FIXED BILLING RATE
            // -------------------------------------------------

            else if (
                billingModel ===
                "Fixed Billing Rate"
            ) {

                const dailyBillRate =
                    billRate / 26;

                basicBilling =
                    dailyBillRate *
                    totalBillableDays;

                serviceCharge =
                    Math.max(
                        0,
                        basicBilling -
                        (
                            payRate / 26 *
                            totalBillableDays
                        )
                    );

                billingAmount =
                    basicBilling;
            }

            // -------------------------------------------------
            // FALLBACK
            // -------------------------------------------------

            else {

                const dailyBillRate =
                    billRate / 26;

                basicBilling =
                    dailyBillRate *
                    totalBillableDays;

                serviceCharge =
                    Math.max(
                        0,
                        basicBilling -
                        (
                            payRate / 26 *
                            totalBillableDays
                        )
                    );

                billingAmount =
                    basicBilling;
            }

            return {

                // ---------------------------------------------
                // ATTENDANCE
                // ---------------------------------------------

                attendance_id:
                    attendance.id,

                deployment_id:
                    deployment.id,

                employee_name:
                    attendance.employee_name ||
                    deployment.candidates?.full_name ||
                    "N/A",

                employee_ref_id:
                    deployment.candidate_id,

                designation:
                    deployment.candidates?.designation ||
                    "N/A",

                billing_month:
                    attendance.billing_month,

                present_days:
                    presentDays,

                absent_days:
                    absentDays,

                leave_days:
                    leaveDays,

                overtime_hours:
                    overtimeHours,

                // ---------------------------------------------
                // CONTRACT
                // ---------------------------------------------

                contract_id:
                    contract.id,

                contract_number:
                    contract.contract_number,

                contract_title:
                    contract.contract_title,

                billing_model:
                    billingModel,

                markup_percentage:
                    Number(
                        contract.markup_percentage || 0
                    ),

                per_head_fee:
                    Number(
                        contract.per_head_fee || 0
                    ),

                credit_terms:
                    contract.credit_terms,

                // ---------------------------------------------
                // RATES
                // ---------------------------------------------

                pay_rate:
                    Number(payRate.toFixed(2)),

                bill_rate:
                    Number(billRate.toFixed(2)),

                // ---------------------------------------------
                // BILLING
                // ---------------------------------------------

                billable_days:
                    totalBillableDays,

                basic_billing:
                    Number(
                        basicBilling.toFixed(2)
                    ),

                service_charge:
                    Number(
                        serviceCharge.toFixed(2)
                    ),

                billing_amount:
                    Number(
                        billingAmount.toFixed(2)
                    ),

                approval_status:
                    attendance.approval_status,

                billing_status:
                    "Ready for Invoice",
            };
        }
    );

    return {
        contract,
        records: records.filter(Boolean),
    };
}

app.get("/api/invoices/:id/payments", async (req, res) => {
    try {
        const invoiceId = getId(req.params.id);

        if (!invoiceId) {

app.post("/api/invoices/:id/payments", async (req, res) => {
    try {

        const invoiceId =
            getId(req.params.id);

        if (!invoiceId) {
            return sendError(
                res,
                400,
                "Invalid invoice ID."
            );
        }

        const {
            amount_received,
            tds_deducted,
            payment_date,
            payment_mode,
            reference_number
        } = req.body;


        // ------------------------------------------------
        // 1. VALIDATE AMOUNT
        // ------------------------------------------------

        if (
            amount_received === undefined ||
            Number(amount_received) <= 0
        ) {
            return sendError(
                res,
                400,
                "amount_received must be a positive number."
            );
        }

        if (!payment_date) {
            return sendError(
                res,
                400,
                "payment_date is required."
            );
        }

        const receivedAmount =
            Number(amount_received);

        const tdsAmount =
            Number(tds_deducted || 0);

        if (tdsAmount < 0) {
            return sendError(
                res,
                400,
                "tds_deducted cannot be negative."
            );
        }


        // ------------------------------------------------
        // 2. GET INVOICE
        // ------------------------------------------------

        const { data: invoice, error: invoiceError } =
            await supabase
                .from("client_billing_invoices")
                .select(`
                    id,
                    total_amount,
                    lifecycle_state,
                    payment_status,
                    due_date
                `)
                .eq("id", invoiceId)
                .maybeSingle();

        if (invoiceError) throw invoiceError;

        if (!invoice) {
            return sendError(
                res,
                404,
                "Invoice not found."
            );
        }


        // ------------------------------------------------
        // 3. INSERT PAYMENT INTO payments_ledger
        // ------------------------------------------------

        const { data: payment, error: insertError } =
            await supabase
                .from("payments_ledger")
                .insert({
                    invoice_id: invoiceId,

                    amount_received:
                        receivedAmount,

                    tds_deducted:
                        tdsAmount,

                    payment_date:
                        payment_date,

                    payment_mode:
                        payment_mode ||
                        "NEFT/RTGS",

                    reference_number:
                        reference_number ||
                        null
                })
                .select()
                .single();

        if (insertError) throw insertError;


        // ------------------------------------------------
        // 4. GET ALL PAYMENTS
        // ------------------------------------------------

        const { data: allPayments, error: sumError } =
            await supabase
                .from("payments_ledger")
                .select(
                    "amount_received, tds_deducted"
                )
                .eq("invoice_id", invoiceId);

        if (sumError) throw sumError;


        // ------------------------------------------------
        // 5. CALCULATE TOTAL RECEIVED
        // ------------------------------------------------

        const totalReceived =
            (allPayments || []).reduce(
                (sum, p) =>
                    sum +
                    Number(
                        p.amount_received || 0
                    ),
                0
            );


        // ------------------------------------------------
        // 6. CALCULATE TOTAL TDS
        // ------------------------------------------------

        const totalTDS =
            (allPayments || []).reduce(
                (sum, p) =>
                    sum +
                    Number(
                        p.tds_deducted || 0
                    ),
                0
            );


        // ------------------------------------------------
        // 7. CALCULATE TOTAL SETTLED
        // ------------------------------------------------

        const totalSettled =
            totalReceived + totalTDS;

        const invoiceTotal =
            Number(invoice.total_amount || 0);

        const balanceRemaining =
            Math.max(
                invoiceTotal -
                totalSettled,
                0
            );


        // ------------------------------------------------
        // 8. DETERMINE PAYMENT STATUS
        // ------------------------------------------------

        let paymentStatus;

        if (balanceRemaining <= 0) {

            // Invoice completely settled
            paymentStatus = "Paid";

        } else if (totalSettled > 0) {

            // Some amount received
            paymentStatus = "Partially Paid";

        } else {

            // No payment received
            const today =
                new Date()
                    .toISOString()
                    .split("T")[0];

            if (
                invoice.due_date &&
                invoice.due_date < today
            ) {

                paymentStatus = "Overdue";

            } else {

                paymentStatus = "Pending";
            }
        }


        // ------------------------------------------------
        // 9. UPDATE ONLY payment_status
        // ------------------------------------------------

        const { data: updatedInvoice, error: updateError } =
            await supabase
                .from("client_billing_invoices")
                .update({
                    payment_status:
                        paymentStatus
                })
                .eq("id", invoiceId)
                .select(`
                    id,
                    lifecycle_state,
                    payment_status,
                    due_date
                `)
                .single();

        if (updateError) {
            throw updateError;
        }


        // ------------------------------------------------
        // 10. RESPONSE
        // ------------------------------------------------

        return res.json({

            success: true,

            message:
                "Payment recorded successfully.",

            payment,

            invoice_id:
                invoiceId,

            invoice_total:
                invoiceTotal,

            total_received:
                totalReceived,

            total_tds:
                totalTDS,

            total_settled:
                totalSettled,

            balance_remaining:
                balanceRemaining,

            payment_status:
                updatedInvoice.payment_status,

            lifecycle_state:
                updatedInvoice.lifecycle_state,

            due_date:
                updatedInvoice.due_date
        });

    } catch (err) {

        console.error(
            "POST /api/invoices/:id/payments:",
            err
        );

        sendError(
            res,
            500,
            err.message
        );
    }
});

                    `)

            if (error) throw error;

            // ---------- NEW: merge in payment totals ----------

            const invoiceIds = (invoices || []).map((inv) => inv.id);

            let paymentTotalsByInvoice = {};

            if (invoiceIds.length > 0) {

                const { data: payments, error: paymentsError } = await supabase
                    .from("payments_ledger")
                    .select("invoice_id, amount_received, tds_deducted")
                    .in("invoice_id", invoiceIds);

                if (paymentsError) throw paymentsError;

                (payments || []).forEach((p) => {
                    const key = p.invoice_id;
                    if (!paymentTotalsByInvoice[key]) {
                        paymentTotalsByInvoice[key] = { total_received: 0, total_tds: 0 };
                    }
                    paymentTotalsByInvoice[key].total_received += Number(p.amount_received || 0);
                    paymentTotalsByInvoice[key].total_tds += Number(p.tds_deducted || 0);
                });
            }

            const enrichedInvoices = (invoices || []).map((inv) => {
                const totals = paymentTotalsByInvoice[inv.id] || { total_received: 0, total_tds: 0 };
                const totalSettled = totals.total_received + totals.total_tds;
                const invoiceTotal = Number(inv.total_amount || 0);

                return {
                    ...inv,
                    total_received: totals.total_received,
                    total_tds: totals.total_tds,
                    total_settled: totalSettled,
                    balance_remaining: Math.max(invoiceTotal - totalSettled, 0)
                };
            });

            // ----------------------------------------------------

            res.json({
                success: true,
                data: enrichedInvoices
            });

        } catch (err) {
            console.error("GET /api/client-billing/invoices:", err);
            sendError(res, 500, err.message);
        }
    }
);

                );
            }

            // -------------------------------------------------
            // GET ATTENDANCE
            // -------------------------------------------------

            const {
                data: attendance,
                error: attendanceError,
            } =
                await supabase
                    .from(
                        "third_party_emp_attendance"
                    )
                    .select(`
                            id,
                            employee_name,
                            client_id,
                            billing_month,
                            status,
                            present_days,
                            absent_days,
                            leave_days,
                            overtime_hours,
                            approval_status,
                            deployment_id
                        `)
                    .eq("id", attendanceId)
                    .single();

            if (attendanceError) {
                throw attendanceError;
            }

            // -------------------------------------------------
            // ONLY APPROVED ATTENDANCE CAN BE BILLED
            // -------------------------------------------------

            if (
                attendance.approval_status !==
                "Approved"
            ) {

                return sendError(
                    res,
                    400,
                    "Only approved attendance can be billed"
                );
            }

            if (!attendance.client_id) {

                return sendError(
                    res,
                    400,
                    "Attendance has no client_id"
                );
            }

            // -------------------------------------------------
            // GET BILLING DATA FOR CLIENT
            // -------------------------------------------------

            const {
                contract,
                records,
            } =
                await getClientBillingData(
                    attendance.client_id,
                    attendance.billing_month
                );

            // -------------------------------------------------
            // FIND CURRENT ATTENDANCE
            // -------------------------------------------------

            const record =
                records.find(
                    (item) =>
                        Number(
                            item.attendance_id
                        ) ===
                        Number(attendanceId)
                );

            if (!record) {

                return sendError(
                    res,
                    404,
                    "Billing record could not be generated"
                );
            }

            res.json({

                success: true,

                data: {

                    contract:
                        contract,

                    billing:
                        record,
                },
            });

        } catch (err) {

            console.error(
                "GET /api/client-billing/:attendanceId:",
                err.message
            );

            sendError(
                res,
                500,
                err.message
            );
        }
    }
);

app.get("/api/job-requirements", async (req, res) => {
    try {
        const { client_id, status } = req.query;

        let query = supabase
            .from("job_requirements")
            .select(`
                    id,
                    client_id,
                    job_title,
                    skills_required,
                    positions_count,
                    location,
                    experience_min,
                    experience_max,
                    salary_lpa,
                    status,
                    created_at
                `)
            .order("created_at", {
                ascending: false
            });

        // Filter by client
        if (client_id) {
            const clientId = Number(client_id);

            if (!Number.isInteger(clientId)) {
                return res.status(400).json({
                    success: false,
                    message: "Invalid client_id"
                });
            }

            query = query.eq(
                "client_id",
                clientId
            );
        }

        // Filter by status
        if (status) {
            query = query.eq(
                "status",
                status
            );
        }

        const {
            data,
            error
        } = await query;

        if (error) {
            throw error;
        }

        return res.json({
            success: true,
            data: data || []
        });

    } catch (err) {

        console.error(
            "GET /api/job-requirements:",
            err
        );

        return res.status(500).json({
            success: false,
            message:
                err.message ||
                "Failed to fetch job requirements"
        });
    }
});

// ============================================================
// 4. ASSIGN CANDIDATE TO JOB REQUIREMENT
//
// POST /api/job-requirements/:id/assign
//
// Body:
//
// {
//     "candidate_id": 10,
//     "client_id": 2
// }
//
// Admin selects an existing candidate.
// No new job requirement is created here.
// ============================================================

app.post(
    "/api/job-requirements/:id/assign",
    async (req, res) => {

        try {

            const jobRequirementId =
                Number(req.params.id);

            const {
                candidate_id,
                client_id
            } = req.body || {};


            // ------------------------------------------------
            // VALIDATE JOB ID
            // ------------------------------------------------

            if (
                !Number.isInteger(
                    jobRequirementId
                )
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "Invalid job requirement ID"
                });
            }


            // ------------------------------------------------
            // VALIDATE CANDIDATE
            // ------------------------------------------------

            const candidateId =
                Number(candidate_id);

            if (
                !Number.isInteger(
                    candidateId
                )
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "Valid candidate_id is required"
                });
            }


            // ------------------------------------------------
            // VALIDATE CLIENT
            // ------------------------------------------------

            const clientId =
                Number(client_id);

            if (
                !Number.isInteger(
                    clientId
                )
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "Valid client_id is required"
                });
            }


            // ------------------------------------------------
            // CHECK JOB REQUIREMENT
            // ------------------------------------------------

            const {
                data: job,
                error: jobError
            } = await supabase
                .from("job_requirements")
                .select(`
                        id,
                        client_id,
                        job_title,
                        positions_count,
                        status
                    `)
                .eq(
                    "id",
                    jobRequirementId
                )
                .single();


            if (jobError) {

                if (
                    jobError.code ===
                    "PGRST116"
                ) {

                    return res.status(404).json({
                        success: false,
                        message:
                            "Job requirement not found"
                    });
                }

                throw jobError;
            }


            // ------------------------------------------------
            // MAKE SURE CLIENT MATCHES JOB
            // ------------------------------------------------

            if (
                Number(job.client_id) !==
                clientId
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "Candidate client does not match job requirement client"
                });
            }


            // ------------------------------------------------
            // CHECK JOB STATUS
            // ------------------------------------------------

            if (
                job.status ===
                "Closed"
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "This job requirement is closed"
                });
            }


            // ------------------------------------------------
            // CHECK CANDIDATE
            // ------------------------------------------------

            const {
                data: candidate,
                error: candidateError
            } = await supabase
                .from("candidates")
                .select(`
                        id,
                        name,
                        full_name,
                        email
                    `)
                .eq(
                    "id",
                    candidateId
                )
                .single();


            if (candidateError) {

                if (
                    candidateError.code ===
                    "PGRST116"
                ) {

                    return res.status(404).json({
                        success: false,
                        message:
                            "Candidate not found"
                    });
                }

                throw candidateError;
            }


            // =================================================
            // IMPORTANT
            // =================================================
            //
            // Your current frontend expects assignment to
            // eventually create/update deployment.
            //
            // If your employee_deployments table has these
            // columns, insert the deployment here.
            //
            // Adjust candidate column names if necessary.
            // =================================================

            const {
                data: deployment,
                error: deploymentError
            } = await supabase
                .from("employee_deployments")
                .insert({

                    client_id:
                        clientId,

                    candidate_id:
                        candidateId,

                    job_requirement_id:
                        jobRequirementId,

                    status:
                        "Active"

                })
                .select()
                .single();


            if (deploymentError) {
                throw deploymentError;
            }


            // =================================================
            // UPDATE JOB STATUS
            //
            // For now:
            //
            // Open → In Progress
            //
            // The requirement can later become Closed when
            // positions_count are completely filled.
            // =================================================

            let newStatus =
                job.status;

            if (
                job.status ===
                "Open"
            ) {

                newStatus =
                    "In Progress";
            }


            const {
                data: updatedJob,
                error: updateError
            } = await supabase
                .from("job_requirements")
                .update({

                    status:
                        newStatus

                })
                .eq(
                    "id",
                    jobRequirementId
                )
                .select()
                .single();


            if (updateError) {
                throw updateError;
            }


            return res.status(201).json({

                success: true,

                message:
                    "Candidate assigned successfully",

                data: {

                    job_requirement:
                        updatedJob,

                    candidate,

                    deployment

                }

            });


        } catch (err) {

            console.error(
                "POST /api/job-requirements/:id/assign:",
                err
            );

            return res.status(500).json({
                success: false,
                message:
                    err.message ||
                    "Failed to assign candidate"
            });
        }
    }
);

const EMPLOYER_STATUTORY_RATE = 0.1525; // 15.25%

function parseCreditTermsDays(creditTerms) {
    if (!creditTerms) return 30;
    const match = String(creditTerms).match(/\d+/);
    return match ? Number(match[0]) : 30;
}

async function generateInvoiceNumber() {
    const { data, error } = await supabase
        .from("client_billing_invoices")
        .select("id")
        .order("id", { ascending: false })
        .limit(1);

    if (error) throw error;

    const nextId = data && data.length > 0 ? data[0].id + 1 : 1;
    return `INV-2026-${String(nextId).padStart(6, "0")}`;
}

// -----------------------------------------------------
// VERIFY PAYMENT CONFIRMATION
// -----------------------------------------------------
app.patch("/api/admin/payment-confirmations/:id/verify", async (req, res) => {
    try {
        const { id } = req.params;

        // Replace this with your actual logged-in admin identity
        const verifiedBy =
            req.body.verified_by ||
            req.headers["x-admin-email"] ||
            "Admin";

        // ---------------------------------------------
        // 1. Get confirmation
        // ---------------------------------------------
        const { data: confirmation, error: confirmationError } =
            await supabase
                .from("payment_confirmations")
                .select("*")
                .eq("id", id)
                .single();

        if (confirmationError || !confirmation) {
            return res.status(404).json({
                success: false,
                message: "Payment confirmation not found",
            });
        }

        // ---------------------------------------------
        // 2. Prevent double verification
        // ---------------------------------------------
        if (confirmation.status === "Verified") {
            return res.status(400).json({
                success: false,
                message: "Payment confirmation is already verified",
            });
        }

        // ---------------------------------------------
        // 3. Prevent rejected confirmation verification
        // ---------------------------------------------
        if (confirmation.status === "Rejected") {
            return res.status(400).json({
                success: false,
                message:
                    "Rejected payment confirmation cannot be verified. Ask the client to report the payment again.",
            });
        }

        // ---------------------------------------------
        // 4. Get invoice
        // ---------------------------------------------
        const { data: invoice, error: invoiceError } =
            await supabase
                .from("client_billing_invoices")
                .select("*")
                .eq("id", confirmation.invoice_id)
                .single();

        if (invoiceError || !invoice) {
            return res.status(404).json({
                success: false,
                message: "Associated invoice not found",
            });
        }

        // ---------------------------------------------
        // 5. Check existing ledger payment
        // ---------------------------------------------
        const { data: existingPayments, error: ledgerError } =
            await supabase
                .from("payments_ledger")
                .select("*")
                .eq("invoice_id", confirmation.invoice_id);

        if (ledgerError) {
            console.error(
                "Ledger lookup error:",
                ledgerError
            );

            return res.status(500).json({
                success: false,
                message: "Unable to check payment ledger",
            });
        }

        const existingPaid = (existingPayments || []).reduce(
            (sum, payment) =>
                sum + Number(payment.amount_received || 0),
            0
        );

        // ---------------------------------------------
        // 6. Prevent overpayment
        // ---------------------------------------------
        const invoiceTotal = Number(invoice.total_amount || 0);

        const amountBeingVerified =
            Number(confirmation.amount_reported || 0);

        const remainingAmount =
            invoiceTotal - existingPaid;

        if (amountBeingVerified <= 0) {
            return res.status(400).json({
                success: false,
                message: "Payment amount must be greater than zero",
            });
        }

        if (amountBeingVerified > remainingAmount) {
            return res.status(400).json({
                success: false,
                message: `Payment exceeds outstanding invoice amount. Outstanding amount is ₹${remainingAmount.toFixed(
                    2
                )}.`,
            });
        }

        // ---------------------------------------------
        // 7. Insert into payments ledger
        // ---------------------------------------------
        const { data: ledgerPayment, error: insertLedgerError } =
            await supabase
                .from("payments_ledger")
                .insert([
                    {
                        invoice_id: confirmation.invoice_id,
                        amount_received: amountBeingVerified,
                        payment_date: confirmation.payment_date,
                        payment_mode: confirmation.payment_mode,
                        reference_number: confirmation.reference_number,

                        // Keep this if your ledger has the field
                        tds_deducted: 0,
                    },
                ])
                .select()
                .single();

        if (insertLedgerError) {
            console.error(
                "Insert payments_ledger error:",
                insertLedgerError
            );

            return res.status(500).json({
                success: false,
                message:
                    "Payment confirmation could not be posted to payment ledger",
            });
        }

        // ---------------------------------------------
        // 8. Mark confirmation Verified
        // ---------------------------------------------
        const { data: updatedConfirmation, error: updateError } =
            await supabase
                .from("payment_confirmations")
                .update({
                    status: "Verified",
                    verified_by: verifiedBy,
                    verified_at: new Date().toISOString(),
                    rejection_reason: null,
                })
                .eq("id", id)
                .select()
                .single();

        if (updateError) {
            console.error(
                "Update confirmation error:",
                updateError
            );

            return res.status(500).json({
                success: false,
                message:
                    "Payment was added to ledger but confirmation status could not be updated",
            });
        }

        // ---------------------------------------------
        // 9. Calculate new invoice payment status
        // ---------------------------------------------
        const newPaidAmount =
            existingPaid + amountBeingVerified;

        const newOutstanding =
            Math.max(invoiceTotal - newPaidAmount, 0);

        let newPaymentStatus = "Pending";

        if (newOutstanding <= 0) {
            newPaymentStatus = "Paid";
        } else if (newPaidAmount > 0) {
            newPaymentStatus = "Partially Paid";
        }

        // ---------------------------------------------
        // 10. Update invoice
        // ---------------------------------------------
        const { error: invoiceUpdateError } =
            await supabase
                .from("client_billing_invoices")
                .update({
                    payment_status: newPaymentStatus,
                })
                .eq("id", confirmation.invoice_id);

        if (invoiceUpdateError) {
            console.error(
                "Invoice status update error:",
                invoiceUpdateError
            );

            // Payment is already in ledger.
            // Don't pretend verification failed.
        }

        // ---------------------------------------------
        // SUCCESS
        // ---------------------------------------------
        res.json({
            success: true,
            message: "Payment verified successfully",
            data: {
                confirmation: updatedConfirmation,
                ledger_payment: ledgerPayment,
                invoice_payment_status: newPaymentStatus,
                amount_paid: newPaidAmount,
                outstanding: newOutstanding,
            },
        });
    } catch (error) {
        console.error(
            "PATCH /api/admin/payment-confirmations/:id/verify:",
            error
        );

        res.status(500).json({
            success: false,
            message: "Failed to verify payment",
        });
    }
});

app.patch("/api/admin/payment-confirmations/:id/reject", async (req, res) => {
    try {
        const { id } = req.params;

        const {
            rejection_reason,
            verified_by,
        } = req.body;

        if (!rejection_reason || !rejection_reason.trim()) {
            return res.status(400).json({
                success: false,
                message: "Rejection reason is required",
            });
        }

        // ---------------------------------------------
        // 1. Get confirmation
        // ---------------------------------------------
        const { data: confirmation, error: confirmationError } =
            await supabase
                .from("payment_confirmations")
                .select("*")
                .eq("id", id)
                .single();

        if (confirmationError || !confirmation) {
            return res.status(404).json({
                success: false,
                message: "Payment confirmation not found",
            });
        }

        // ---------------------------------------------
        // 2. Prevent changing verified payment
        // ---------------------------------------------
        if (confirmation.status === "Verified") {
            return res.status(400).json({
                success: false,
                message:
                    "A verified payment cannot be rejected",
            });
        }

        // ---------------------------------------------
        // 3. Reject
        // ---------------------------------------------
        const { data: updatedConfirmation, error: updateError } =
            await supabase
                .from("payment_confirmations")
                .update({
                    status: "Rejected",
                    rejection_reason: rejection_reason.trim(),
                    verified_by:
                        verified_by ||
                        req.headers["x-admin-email"] ||
                        "Admin",
                    verified_at: new Date().toISOString(),
                })
                .eq("id", id)
                .select()
                .single();

        if (updateError) {
            console.error(
                "Reject payment confirmation:",
                updateError
            );

            return res.status(500).json({
                success: false,
                message: "Failed to reject payment confirmation",
            });
        }

        // IMPORTANT:
        // Rejection does NOT create a ledger entry.
        // Rejection does NOT increase amount paid.

        res.json({
            success: true,
            message: "Payment confirmation rejected",
            data: updatedConfirmation,
        });
    } catch (error) {
        console.error(
            "PATCH /api/admin/payment-confirmations/:id/reject:",
            error
        );

        res.status(500).json({
            success: false,
            message: "Failed to reject payment confirmation",
        });
    }
});

            // -------------------------------------------------

                return res.json({
                    success: true,
                    disputes: [],
                });
            }

            // -------------------------------------------------
            // GET INVOICE IDS
            // -------------------------------------------------

            const invoiceIds = [
                ...new Set(
                    disputes.map(
                        (dispute) => dispute.invoice_id
                    )

                );

                return res.status(500).json({
                    success: false,
                    message: invoiceError.message,
                });

            // GET CLIENT IDS
            // -------------------------------------------------

                ...new Set(
                    disputes.map(
                        (dispute) => dispute.client_id
                    )
                ),
            ];

                    invoice.id,

                            dispute.client_id,

                        client_name:
                            client?.company_name ||
                            `Client #${dispute.client_id}`,

                        dispute_type:
                            dispute.dispute_type,

                        reviewed_by:
                            dispute.reviewed_by,

                        reviewed_at:
                            dispute.reviewed_at,

                                invoice?.total_amount || 0
                            ),

                        payment_status:
                            invoice?.payment_status ||
                            "Pending",

                        lifecycle_state:

                            null,

                        due_date:
                            invoice?.due_date ||
                            null,

                        billing_month:
                            invoice?.billing_month ||
                            null,
                    };
                });

            return res.json({
                success: true,
                disputes: formattedDisputes,

            const allowedStatuses = [
                "Accepted",
                "Rejected",
                "Resolved",
            ];

            if (
                !allowedStatuses.includes(status)
            ) {
                return res.status(400).json({
                    success: false,
                    message:
                        "Invalid dispute status.",
                });
            }

            // -------------------------------------------------
            // VALIDATE ADMIN
            // -------------------------------------------------

            if (
                !reviewed_by ||
                !String(reviewed_by).trim()
            ) {
                return res.status(400).json({
                    success: false,
                    message:
                        "reviewed_by is required.",
                });
            }

            // -------------------------------------------------
            // RESPONSE REQUIRED
            // -------------------------------------------------

            if (
                !admin_response ||
                !String(admin_response).trim()
            ) {
                return res.status(400).json({
                    success: false,
                    message:
                        "Admin response is required.",
                });
            }

            // -------------------------------------------------
            // FIND DISPUTE
            // -------------------------------------------------

            const {
                data: dispute,
                error: disputeFetchError,
            } = await supabase
                .from("client_invoice_disputes")
                .select(`
          id,
          invoice_id,
          client_id,
          dispute_type,
          reason,
          status,
          submitted_at
        `)
                .eq("id", disputeId)
                .single();

            if (
                disputeFetchError ||
                !dispute
            ) {
                return res.status(404).json({
                    success: false,
                    message:
                        "Invoice dispute not found.",
                });
            }

            // -------------------------------------------------
            // ONLY PENDING DISPUTES CAN BE REVIEWED
            // -------------------------------------------------

            if (
                dispute.status !==
                "Pending Review"
            ) {
                return res.status(409).json({
                    success: false,
                    message:
                        `This dispute is already ${dispute.status}.`,
                });
            }

            // -------------------------------------------------
            // UPDATE DISPUTE
            // -------------------------------------------------

            const {
                data: updatedDispute,
                error: updateError,
            } = await supabase
                .from("client_invoice_disputes")
                .update({
                    status,

                    reviewed_by:
                        String(
                            reviewed_by
                        ).trim(),

                    reviewed_at:
                        new Date().toISOString(),

                    admin_response:
                        String(
                            admin_response
                        ).trim(),
                })
                .eq(
                    "id",
                    disputeId
                )
                .select()
                .single();

            if (updateError) {
                console.error(
                    "Update dispute error:",
                    updateError
                );

                return res.status(500).json({
                    success: false,
                    message:
                        updateError.message,
                });
            }

            // -------------------------------------------------
            // RESPONSE
            // -------------------------------------------------

            return res.json({
                success: true,

                message:
                    `Invoice dispute marked as ${status}.`,

                dispute:
                    updatedDispute,
            });

        } catch (error) {

            console.error(
                "Review invoice dispute error:",
                error
            );

            return res.status(500).json({
                success: false,
                message:
                    "Server error reviewing invoice dispute.",
            });
        }
    }
);

// =====================================================
// ADMIN PAYMENT CONFIRMATIONS
// GET /api/payments
// =====================================================

app.get("/api/payments", async (req, res) => {
  try {

    // ===============================================
    // PAYMENT CONFIRMATIONS
    // ===============================================

    const {
      data: confirmations,
      error: confirmationError
    } = await supabase
      .from("payment_confirmations")
      .select(`
        id,
        invoice_id,
        client_id,
        amount_reported,
        payment_date,
        payment_mode,
        reference_number,
        payment_proof_url,
        notes,
        status,
        submitted_at,
        verified_by,
        verified_at,
        rejection_reason
      `)
      .order("submitted_at", {
        ascending: false
      });

    if (confirmationError) {
      throw confirmationError;
    }

    if (!confirmations || confirmations.length === 0) {
      return res.json({
        success: true,
        payments: []
      });
    }

    // ===============================================
    // IDS
    // ===============================================

    const invoiceIds = [
      ...new Set(confirmations.map(p => p.invoice_id))
    ];

    const clientIds = [
      ...new Set(confirmations.map(p => p.client_id))
    ];

    // ===============================================
    // INVOICES
    // ===============================================

    const {
      data: invoices,
      error: invoiceError
    } = await supabase
      .from("client_billing_invoices")
      .select(`
        id,
        invoice_number,
        client_id,
        total_amount,
        payment_status,
        billing_month,
        due_date
      `)
      .in("id", invoiceIds);

    if (invoiceError) {
      throw invoiceError;
    }

    // ===============================================
    // CLIENTS
    // ===============================================

    const {
      data: clients,
      error: clientError
    } = await supabase
      .from("clients")
      .select(`
        id,
        company_name
      `)
      .in("id", clientIds);

    if (clientError) {
      throw clientError;
    }

    // ===============================================
    // VERIFIED PAYMENT LEDGER
    // ===============================================

    const {
      data: ledger,
      error: ledgerError
    } = await supabase
      .from("payments_ledger")
      .select(`
        id,
        invoice_id,
        amount_received,
        tds_deducted,
        payment_date
      `)
      .in("invoice_id", invoiceIds);

    if (ledgerError) {
      throw ledgerError;
    }

    // ===============================================
    // BUILD MAPS
    // ===============================================

    const invoiceMap = {};

    (invoices || []).forEach(inv => {
      invoiceMap[inv.id] = inv;
    });

    const clientMap = {};

    (clients || []).forEach(client => {
      clientMap[client.id] = client;
    });

    const ledgerMap = {};

    (ledger || []).forEach(item => {

      if (!ledgerMap[item.invoice_id]) {
        ledgerMap[item.invoice_id] = {
          amountPaid: 0,
          tds: 0
        };
      }

      ledgerMap[item.invoice_id].amountPaid +=
        Number(item.amount_received || 0);

      ledgerMap[item.invoice_id].tds +=
        Number(item.tds_deducted || 0);
    });

    // ===============================================
    // BUILD RESPONSE
    // ===============================================

    const result = confirmations.map(payment => {

      const invoice =
        invoiceMap[payment.invoice_id] || {};

      const client =
        clientMap[payment.client_id] || {};

      const paymentInfo =
        ledgerMap[payment.invoice_id] || {
          amountPaid: 0,
          tds: 0
        };

      const totalAmount =
        Number(invoice.total_amount || 0);

      const amountPaid =
        Number(paymentInfo.amountPaid || 0);

      const tds =
        Number(paymentInfo.tds || 0);

      const settled =
        amountPaid + tds;

      const outstanding =
        Math.max(0, totalAmount - settled);

      let completionStatus = "Pending";

      if (outstanding <= 0) {
        completionStatus = "Paid";
      }
      else if (settled > 0) {
        completionStatus = "Partially Paid";
      }
      else {
        completionStatus = "Pending";
      }

      return {

        ...payment,

        invoice_number:
          invoice.invoice_number,

        invoice_amount:
          totalAmount,

        client_name:
          client.company_name,

        amount_paid:
          amountPaid,

        tds_deducted:
          tds,

        amount_settled:
          settled,

        outstanding,

        payment_completion_status:
          completionStatus,

        invoice_status:
          invoice.payment_status
      };

    });

    return res.json({
      success: true,
      payments: result
    });

  }
  catch (error) {

    console.error(
      "GET /api/payments error:",
      error
    );

    return res.status(500).json({
      success: false,
      error: error.message
    });

  }
});

// 2. VERIFY PAYMENT CONFIRMATION
// PATCH /api/payments/:confirmationId/verify
// =====================================================

app.patch(
    "/api/payments/:confirmationId/verify",
    async (req, res) => {
        try {
            const confirmationId = Number(
                req.params.confirmationId
            );

            const {
                verified_by,
                tds_deducted = 0
            } = req.body;

            // =====================================================
            // VALIDATION
            // =====================================================

            if (
                !Number.isInteger(confirmationId) ||
                confirmationId <= 0
            ) {
                return res.status(400).json({
                    success: false,
                    message: "Valid confirmation ID is required."
                });
            }

            if (
                !verified_by ||
                !String(verified_by).trim()
            ) {
                return res.status(400).json({
                    success: false,
                    message: "verified_by is required."
                });
            }

            const tds = Number(tds_deducted);

            if (
                !Number.isFinite(tds) ||
                tds < 0
            ) {
                return res.status(400).json({
                    success: false,
                    message: "Invalid TDS amount."
                });
            }

            // =====================================================
            // GET PAYMENT CONFIRMATION
            // =====================================================

            const {
                data: confirmation,
                error: confirmationError
            } = await supabase
                .from("payment_confirmations")
                .select(`
                    id,
                    invoice_id,
                    client_id,
                    amount_reported,
                    payment_date,
                    payment_mode,
                    reference_number,
                    status
                `)
                .eq("id", confirmationId)
                .single();

            if (
                confirmationError ||
                !confirmation
            ) {
                return res.status(404).json({
                    success: false,
                    message: "Payment confirmation not found."
                });
            }

            // =====================================================
            // PAYMENT MUST BE PENDING
            // =====================================================

            if (
                confirmation.status !==
                "Pending Verification"
            ) {
                return res.status(409).json({
                    success: false,
                    message:
                        `This payment confirmation is already ${confirmation.status}.`
                });
            }

            // =====================================================
            // GET INVOICE
            // =====================================================

            const {
                data: invoice,
                error: invoiceError
            } = await supabase
                .from("client_billing_invoices")
                .select(`
                    id,
                    invoice_number,
                    client_id,
                    total_amount,
                    payment_status,
                    lifecycle_state
                `)
                .eq(
                    "id",
                    confirmation.invoice_id
                )
                .single();

            if (
                invoiceError ||
                !invoice
            ) {
                return res.status(404).json({
                    success: false,
                    message: "Invoice not found."
                });
            }

            // =====================================================
            // CHECK CLIENT OWNERSHIP
            // =====================================================

            if (
                Number(invoice.client_id) !==
                Number(confirmation.client_id)
            ) {
                return res.status(400).json({
                    success: false,
                    message:
                        "Payment confirmation does not belong to this invoice."
                });
            }

            // =====================================================
            // CHECK ACTIVE INVOICE DISPUTE
            // =====================================================

            const {
                data: activeDisputes,
                error: disputeError
            } = await supabase
                .from("client_invoice_disputes")
                .select(`
                    id,
                    status
                `)
                .eq(
                    "invoice_id",
                    invoice.id
                )
                .eq(
                    "status",
                    "Pending Review"
                )
                .limit(1);

            if (disputeError) {
                console.error(
                    "Dispute check error:",
                    disputeError
                );

                return res.status(500).json({
                    success: false,
                    message:
                        "Unable to check invoice dispute status."
                });
            }

            if (
                activeDisputes &&
                activeDisputes.length > 0
            ) {
                return res.status(409).json({
                    success: false,
                    message:
                        "Payment verification is disabled because this invoice has an unresolved dispute.",
                    dispute_id:
                        activeDisputes[0].id
                });
            }

            // =====================================================
            // PAYMENT AMOUNT
            // =====================================================

            const amountReceived = Number(
                confirmation.amount_reported
            );

            if (
                !Number.isFinite(amountReceived) ||
                amountReceived <= 0
            ) {
                return res.status(400).json({
                    success: false,
                    message: "Invalid payment amount."
                });
            }

            // =====================================================
            // GET EXISTING LEDGER PAYMENTS
            // =====================================================

            const {
                data: existingPayments,
                error: ledgerFetchError
            } = await supabase
                .from("payments_ledger")
                .select(`
                    id,
                    amount_received,
                    tds_deducted
                `)
                .eq(
                    "invoice_id",
                    confirmation.invoice_id
                );

            if (ledgerFetchError) {
                console.error(
                    "Ledger fetch error:",
                    ledgerFetchError
                );

                return res.status(500).json({
                    success: false,
                    message:
                        ledgerFetchError.message
                });
            }

            // =====================================================
            // CALCULATE EXISTING TOTALS
            // =====================================================

            const alreadyReceived =
                (existingPayments || []).reduce(
                    (sum, payment) =>
                        sum +
                        Number(
                            payment.amount_received || 0
                        ),
                    0
                );

            const alreadyTds =
                (existingPayments || []).reduce(
                    (sum, payment) =>
                        sum +
                        Number(
                            payment.tds_deducted || 0
                        ),
                    0
                );

            // =====================================================
            // INVOICE TOTAL
            // =====================================================

            const totalAmount = Number(
                invoice.total_amount || 0
            );

            if (
                !Number.isFinite(totalAmount) ||
                totalAmount <= 0
            ) {
                return res.status(400).json({
                    success: false,
                    message:
                        "Invalid invoice total amount."
                });
            }

            // =====================================================
            // CURRENT SETTLEMENT
            // =====================================================

            const currentSettlement =
                amountReceived + tds;

            const currentOutstanding =
                Math.max(
                    0,
                    totalAmount -
                    alreadyReceived -
                    alreadyTds
                );

            // =====================================================
            // PREVENT OVERPAYMENT
            // =====================================================

            if (
                currentSettlement >
                currentOutstanding + 0.01
            ) {
                return res.status(400).json({
                    success: false,
                    message:
                        `Payment exceeds outstanding invoice amount. Outstanding: ₹${currentOutstanding.toFixed(2)}`
                });
            }

            // =====================================================
            // MARK PAYMENT CONFIRMATION AS VERIFIED
            // =====================================================

            const {
                data: updatedConfirmationRows,
                error: confirmationUpdateError
            } = await supabase
                .from("payment_confirmations")
                .update({
                    status: "Verified",

                    verified_by:
                        String(
                            verified_by
                        ).trim(),

                    verified_at:
                        new Date().toISOString(),

                    rejection_reason: null
                })
                .eq(
                    "id",
                    confirmationId
                )
                .eq(
                    "status",
                    "Pending Verification"
                )
                .select();

            if (confirmationUpdateError) {
                console.error(
                    "Confirmation update error:",
                    confirmationUpdateError
                );

                return res.status(500).json({
                    success: false,
                    message:
                        confirmationUpdateError.message
                });
            }

            // =====================================================
            // RACE CONDITION PROTECTION
            // =====================================================

            if (
                !updatedConfirmationRows ||
                updatedConfirmationRows.length === 0
            ) {
                return res.status(409).json({
                    success: false,
                    message:
                        "This payment confirmation was already processed by someone else. Please refresh."
                });
            }

            const updatedConfirmation =
                updatedConfirmationRows[0];

            // =====================================================
            // INSERT PAYMENT INTO LEDGER
            // =====================================================

            const {
                data: ledgerPayment,
                error: ledgerInsertError
            } = await supabase
                .from("payments_ledger")
                .insert({
                    invoice_id:
                        confirmation.invoice_id,

                    amount_received:
                        amountReceived,

                    payment_date:
                        confirmation.payment_date,

                    payment_mode:
                        confirmation.payment_mode,

                    reference_number:
                        confirmation.reference_number,

                    tds_deducted:
                        tds
                })
                .select()
                .single();

            // =====================================================
            // ROLLBACK CONFIRMATION IF LEDGER INSERT FAILS
            // =====================================================

            if (ledgerInsertError) {
                console.error(
                    "Ledger insert error:",
                    ledgerInsertError
                );

                await supabase
                    .from("payment_confirmations")
                    .update({
                        status:
                            "Pending Verification",

                        verified_by: null,

                        verified_at: null
                    })
                    .eq(
                        "id",
                        confirmationId
                    );

                return res.status(500).json({
                    success: false,
                    message:
                        ledgerInsertError.message
                });
            }

            // =====================================================
            // NEW PAYMENT TOTALS
            // =====================================================

            const newTotalReceived =
                alreadyReceived +
                amountReceived;

            const newTotalTds =
                alreadyTds +
                tds;

            const totalSettled =
                newTotalReceived +
                newTotalTds;

            const outstandingAmount =
                Math.max(
                    0,
                    totalAmount -
                    totalSettled
                );

            // =====================================================
            // DETERMINE PAYMENT STATUS
            // =====================================================

            const invoiceStatus =
                totalSettled >=
                totalAmount - 0.01
                    ? "Paid"
                    : "Partially Paid";

            // =====================================================
            // DETERMINE LIFECYCLE STATE
            // =====================================================

            let lifecycleState =
                invoice.lifecycle_state;

            if (
                invoiceStatus === "Paid"
            ) {
                lifecycleState = "Paid";
            } else if (
                invoiceStatus === "Partially Paid"
            ) {
                lifecycleState = "Partially Paid";
            }

            // =====================================================
            // UPDATE INVOICE
            // =====================================================

            const {
                data: updatedInvoice,
                error: invoiceUpdateError
            } = await supabase
                .from("client_billing_invoices")
                .update({
                    payment_status:
                        invoiceStatus,

                    lifecycle_state:
                        lifecycleState
                })
                .eq(
                    "id",
                    confirmation.invoice_id
                )
                .select()
                .single();

            if (invoiceUpdateError) {
                console.error(
                    "Invoice update error:",
                    invoiceUpdateError
                );

                // ---------------------------------------------
                // ROLLBACK LEDGER ENTRY
                // ---------------------------------------------

                await supabase
                    .from("payments_ledger")
                    .delete()
                    .eq(
                        "id",
                        ledgerPayment.id
                    );

                // ---------------------------------------------
                // ROLLBACK PAYMENT CONFIRMATION
                // ---------------------------------------------

                await supabase
                    .from("payment_confirmations")
                    .update({
                        status:
                            "Pending Verification",

                        verified_by: null,

                        verified_at: null
                    })
                    .eq(
                        "id",
                        confirmationId
                    );

                return res.status(500).json({
                    success: false,
                    message:
                        "Payment verification failed while updating the invoice."
                });
            }

            // =====================================================
            // SUCCESS RESPONSE
            // =====================================================

            return res.json({
                success: true,

                message:
                    "Payment verified successfully.",

                confirmation:
                    updatedConfirmation,

                ledger_payment:
                    ledgerPayment,

                invoice: {
                    id:
                        updatedInvoice.id,

                    invoice_number:
                        updatedInvoice.invoice_number,

                    payment_status:
                        invoiceStatus,

                    lifecycle_state:
                        updatedInvoice.lifecycle_state,

                    total_amount:
                        totalAmount,

                    amount_received:
                        newTotalReceived,

                    tds_deducted:
                        newTotalTds,

                    outstanding:
                        outstandingAmount
                }
            });

        } catch (error) {
            console.error(
                "Verify payment exception:",
                error
            );

            return res.status(500).json({
                success: false,
                message:
                    "Server error verifying payment.",
                error:
                    error.message
            });
        }
    }
);

// 3. REJECT PAYMENT CONFIRMATION
// PATCH /api/payments/:confirmationId/reject
// =====================================================

app.patch(
    "/api/payments/:confirmationId/reject",
    async (req, res) => {

        try {

            const confirmationId =
                Number(
                    req.params.confirmationId
                );

            const {
                verified_by,
                rejection_reason
            } = req.body;

            // -----------------------------------------
            // VALIDATION
            // -----------------------------------------

            if (
                !Number.isInteger(
                    confirmationId
                ) ||
                confirmationId <= 0
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "Valid confirmation ID is required."
                });
            }

            if (
                !verified_by ||
                !String(verified_by).trim()
            ) {

                return res.status(400).json({
                    success: false,
                    message:
                        "verified_by is required."
                });
            }

            const reason =
                String(
                    rejection_reason || ""
                ).trim();

            if (reason.length < 5) {

                return res.status(400).json({
                    success: false,
                    message:
                        "Please provide a rejection reason."
                });
            }

            // -----------------------------------------
            // GET CONFIRMATION
            // -----------------------------------------

            const {
                data: confirmation,
                error: fetchError
            } = await supabase
                .from("payment_confirmations")
                .select(`
                    id,
                    invoice_id,
                    status
                `)
                .eq(
                    "id",
                    confirmationId
                )
                .single();

            if (
                fetchError ||
                !confirmation
            ) {

                return res.status(404).json({
                    success: false,
                    message:
                        "Payment confirmation not found."
                });
            }

            if (
                confirmation.status !==
                "Pending Verification"
            ) {

                return res.status(409).json({
                    success: false,
                    message:
                        `This payment confirmation is already ${confirmation.status}.`
                });
            }

            // -----------------------------------------
            // CHECK ACTIVE INVOICE DISPUTE
            // -----------------------------------------

            const {
                data: activeDisputes,
                error: disputeError
            } = await supabase
                .from("client_invoice_disputes")
                .select(`
                    id,
                    status
                `)
                .eq(
                    "invoice_id",
                    confirmation.invoice_id
                )
                .eq(
                    "status",
                    "Pending Review"
                )
                .limit(1);

            if (disputeError) {

                console.error(
                    "Dispute check error:",
                    disputeError
                );

                return res.status(500).json({
                    success: false,
                    message:
                        "Unable to check invoice dispute status."
                });
            }

            if (
                activeDisputes &&
                activeDisputes.length > 0
            ) {

                return res.status(409).json({
                    success: false,
                    message:
                        "Payment rejection is disabled because this invoice has an unresolved dispute.",
                    dispute_id:
                        activeDisputes[0].id
                });
            }

            // -----------------------------------------
            // REJECT PAYMENT
            // -----------------------------------------

            const {
                data: rejectedRows,
                error: updateError
            } = await supabase
                .from("payment_confirmations")
                .update({

                    status:
                        "Rejected",

                    verified_by:
                        String(
                            verified_by
                        ).trim(),

                    verified_at:
                        new Date().toISOString(),

                    rejection_reason:
                        reason

                })
                .eq(
                    "id",
                    confirmationId
                )
                .eq(
                    "status",
                    "Pending Verification"
                )
                .select();

            if (updateError) {

                console.error(
                    "Reject payment error:",
                    updateError
                );

                return res.status(500).json({
                    success: false,
                    message:
                        updateError.message
                });
            }

            if (
                !rejectedRows ||
                rejectedRows.length === 0
            ) {

                return res.status(409).json({
                    success: false,
                    message:
                        "This payment confirmation was already processed by someone else. Please refresh."
                });
            }

            // -----------------------------------------
            // IMPORTANT
            // -----------------------------------------
            // Rejected payments are NOT inserted
            // into payments_ledger.
            // -----------------------------------------

            return res.json({

                success: true,

                message:
                    "Payment confirmation rejected.",

                confirmation:
                    rejectedRows[0]

            });

        } catch (error) {

            console.error(
                "Reject payment exception:",
                error
            );

            return res.status(500).json({
                success: false,
                message:
                    "Server error rejecting payment.",
                error: error.message
            });
        }

    }
);

// =====================================================
// ADMIN INVOICE DISPUTES
// GET /api/invoice-disputes
// ONLY PENDING DISPUTES
// =====================================================

app.get("/api/invoice-disputes", async (req, res) => {
  try {

    const {
      data: disputes,
      error
    } = await supabase
      .from("client_invoice_disputes")
      .select(`
        id,
        invoice_id,
        client_id,
        dispute_type,
        reason,
        status,
        submitted_at,
        reviewed_by,
        reviewed_at,
        admin_response
      `)
      .eq("status", "Pending Review")
      .order("submitted_at", {
        ascending: false
      });

    if (error) {
      throw error;
    }

    if (!disputes || disputes.length === 0) {

      return res.json({
        success: true,
        disputes: []
      });

    }

    const invoiceIds = [
      ...new Set(disputes.map(d => d.invoice_id))
    ];

    const clientIds = [
      ...new Set(disputes.map(d => d.client_id))
    ];

    // ===============================================
    // FETCH INVOICES
    // ===============================================

    const {
      data: invoices
    } = await supabase
      .from("client_billing_invoices")
      .select(`
        id,
        invoice_number,
        total_amount
      `)
      .in("id", invoiceIds);

    // ===============================================
    // FETCH CLIENTS
    // ===============================================

    const {
      data: clients
    } = await supabase
      .from("clients")
      .select(`
        id,
        company_name
      `)
      .in("id", clientIds);

    const invoiceMap = {};
    const clientMap = {};

    (invoices || []).forEach(i => {
      invoiceMap[i.id] = i;
    });

    (clients || []).forEach(c => {
      clientMap[c.id] = c;
    });

    const result = disputes.map(dispute => ({
      ...dispute,

      invoice_number:
        invoiceMap[dispute.invoice_id]
          ?.invoice_number || "-",

      invoice_amount:
        Number(
          invoiceMap[dispute.invoice_id]
            ?.total_amount || 0
        ),

      client_name:
        clientMap[dispute.client_id]
          ?.company_name || "-"
    }));

    return res.json({
      success: true,
      disputes: result
    });

  }
  catch (error) {

    console.error(
      "Invoice disputes error:",
      error
    );

    return res.status(500).json({
      success: false,
      error: error.message
    });

  }
});

// =====================================================
// VERIFY PAYMENT
// PATCH /api/payments/:id/verify
// =====================================================

app.patch(
  "/api/payments/:id/verify",
  async (req, res) => {

    try {

      const paymentId =
        Number(req.params.id);

      const {
        verified_by,
        tds_deducted
      } = req.body;

      // ==========================================
      // GET PAYMENT
      // ==========================================

      const {
        data: payment,
        error: paymentError
      } = await supabase
        .from("payment_confirmations")
        .select("*")
        .eq("id", paymentId)
        .single();

      if (paymentError || !payment) {

        return res.status(404).json({
          success: false,
          error: "Payment confirmation not found"
        });

      }

      if (
        String(payment.status)
          .toLowerCase() === "verified"
      ) {

        return res.status(400).json({
          success: false,
          error: "Payment already verified"
        });

      }

      // ==========================================
      // UPDATE CONFIRMATION
      // ==========================================

      const {
        error: updateError
      } = await supabase
        .from("payment_confirmations")
        .update({
          status: "Verified",
          verified_by: verified_by || "Admin",
          verified_at: new Date().toISOString(),
          rejection_reason: null
        })
        .eq("id", paymentId);

      if (updateError) {
        throw updateError;
      }

      // ==========================================
      // INSERT LEDGER
      // ==========================================

      const {
        error: ledgerError
      } = await supabase
        .from("payments_ledger")
        .insert({

          invoice_id:
            payment.invoice_id,

          amount_received:
            payment.amount_reported,

          payment_date:
            payment.payment_date,

          payment_mode:
            payment.payment_mode,

          reference_number:
            payment.reference_number,

          tds_deducted:
            Number(tds_deducted || 0)

        });

      if (ledgerError) {
        throw ledgerError;
      }

      return res.json({
        success: true,
        message: "Payment verified successfully"
      });

    }
    catch (error) {

      console.error(
        "Verify payment error:",
        error
      );

      return res.status(500).json({
        success: false,
        error: error.message
      });

    }

  }
);

// =====================================================
// REJECT PAYMENT
// PATCH /api/payments/:id/reject
// =====================================================

app.patch(
  "/api/payments/:id/reject",
  async (req, res) => {

    try {

      const paymentId =
        Number(req.params.id);

      const {
        verified_by,
        rejection_reason
      } = req.body;

      if (
        !rejection_reason ||
        rejection_reason.trim().length < 5
      ) {

        return res.status(400).json({
          success: false,
          error: "Rejection reason is required"
        });

      }

      const {
        error
      } = await supabase
        .from("payment_confirmations")
        .update({

          status: "Rejected",

          verified_by:
            verified_by || "Admin",

          verified_at:
            new Date().toISOString(),

          rejection_reason:
            rejection_reason.trim()

        })
        .eq("id", paymentId);

      if (error) {
        throw error;
      }

      return res.json({
        success: true,
        message:
          "Payment rejected successfully. Client may submit a new payment."
      });

    }
    catch (error) {

      console.error(error);

      return res.status(500).json({
        success: false,
        error: error.message
      });

    }

  }
);

// =====================================================
// RESOLVE INVOICE DISPUTE
// PATCH /api/invoice-disputes/:id/resolve
// =====================================================

app.patch("/api/invoice-disputes/:id/resolve", async (req, res) => {
    try {
        const disputeId = Number(req.params.id);

        const {
            admin_response,
            reviewed_by = "Admin",
        } = req.body;

        if (!Number.isInteger(disputeId)) {
            return res.status(400).json({
                success: false,
                message: "Invalid dispute ID.",
            });
        }

        if (
            !admin_response ||
            admin_response.trim().length < 5
        ) {
            return res.status(400).json({
                success: false,
                message: "Admin response is required.",
            });
        }

        // -------------------------------------------------
        // Get dispute
        // -------------------------------------------------

        const { data: dispute, error: fetchError } =
            await supabase
                .from("client_invoice_disputes")
                .select(`
                    id,
                    invoice_id,
                    client_id,
                    status
                `)
                .eq("id", disputeId)
                .single();

        if (fetchError) {
            console.error(
                "Fetch dispute error:",
                fetchError
            );

            return res.status(500).json({
                success: false,
                message: fetchError.message,
            });
        }

        if (!dispute) {
            return res.status(404).json({
                success: false,
                message: "Invoice dispute not found.",
            });
        }

        // -------------------------------------------------
        // Only pending disputes can be resolved
        // Supports both DB values:
        // Pending
        // Pending Review
        // -------------------------------------------------

        const currentStatus =
            String(dispute.status || "")
                .trim()
                .toLowerCase();

        if (
            currentStatus !== "pending" &&
            currentStatus !== "pending review"
        ) {
            return res.status(409).json({
                success: false,
                message:
                    "Only pending invoice disputes can be resolved.",
            });
        }

        // -------------------------------------------------
        // Resolve dispute
        // -------------------------------------------------

        const { data: updatedDispute, error: updateError } =
            await supabase
                .from("client_invoice_disputes")
                .update({
                    status: "Verified",
                    reviewed_by: reviewed_by,
                    reviewed_at: new Date().toISOString(),
                    admin_response:
                        admin_response.trim(),
                })
                .eq("id", disputeId)
                .in("status", [
                    "Pending",
                    "Pending Review",
                ])
                .select(`
                    id,
                    invoice_id,
                    client_id,
                    dispute_type,
                    reason,
                    status,
                    submitted_at,
                    reviewed_by,
                    reviewed_at,
                    admin_response
                `)
                .single();

        if (updateError) {
            console.error(
                "Resolve dispute update error:",
                updateError
            );

            return res.status(500).json({
                success: false,
                message:
                    updateError.message,
            });
        }

        if (!updatedDispute) {
            return res.status(409).json({
                success: false,
                message:
                    "The dispute could not be resolved. It may have already been processed.",
            });
        }

        return res.json({
            success: true,
            message:
                "Invoice dispute resolved successfully. Client can now make payment.",
            dispute: updatedDispute,
        });

    } catch (error) {

        console.error(
            "Resolve invoice dispute error:",
            error
        );

        return res.status(500).json({
            success: false,
            message:
                error.message ||
                "Failed to resolve invoice dispute.",
        });
    }
});
